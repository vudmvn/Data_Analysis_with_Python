\section{Tổng quan về Tiền xử lý Dữ liệu}

\begin{frame}{Động lực của bài học}
Trong thực tế, dữ liệu thu thập được hiếm khi sẵn sàng để đưa ngay vào phân tích hay mô hình học máy:
\begin{itemize}
  \item \textbf{Giá trị bị thiếu (Missing values):} do người dùng bỏ trống hoặc lỗi truyền nhận.
  \item \textbf{Giá trị bất thường (Outliers):} có thể là lỗi nhập liệu hoặc khách hàng đặc biệt.
  \item \textbf{Định dạng chưa đúng:} số được lưu thành chuỗi, ngày tháng sai định dạng.
  \item \textbf{Biến phân loại (Categorical variables):} dạng chuỗi cần chuyển thành dạng số.
  \item \textbf{Thang đo chênh lệch (Scale difference):} thu nhập hàng chục triệu vs.\ tuổi dưới 100.
  \item \textbf{Mất cân bằng lớp (Class imbalance):} tỷ lệ gian lận hoặc bệnh tật rất thấp.
\end{itemize}

\smallgap
\begin{alertblock}{Nguyên tắc cơ bản}
\textbf{Garbage In, Garbage Out:} Một mô hình phức tạp huấn luyện trên dữ liệu bẩn sẽ cho ra kết quả sai lệch và không đáng tin cậy.
\end{alertblock}
\end{frame}

  \item Giảm tác động tiêu cực của nhiễu.

\end{itemize}

\begin{frame}{Vì sao thuật toán cần Tiền xử lý?}
Nhiều thuật toán học máy rất nhạy cảm với thang đo của biến:
\begin{itemize}
  \item Giả sử có hai biến: \code{Income} (5.000.000 -- 100.000.000) và \code{Age} (18 -- 70).
  \item Khi dùng khoảng cách Euclidean:
  \[
  d = \sqrt{(\text{Income}_1 - \text{Income}_2)^2 + (\text{Age}_1 - \text{Age}_2)^2}
  \]
  \item Biến \code{Income} với độ lớn vượt trội sẽ \textbf{chi phối gần như toàn bộ} khoảng cách, khiến biến \code{Age} mất hoàn toàn ý nghĩa!
\end{itemize}

\smallgap
\textbf{Các thuật toán chịu ảnh hưởng mạnh:}
\begin{itemize}
  \item k-Nearest Neighbors (k-NN), k-Means Clustering
  \item Support Vector Machines (SVM)
  \item Hồi quy có điều chuẩn (Ridge, Lasso Logistic/Linear Regression)
  \item Mạng nơ-ron nhân tạo (Neural Networks)
\end{itemize}
\end{frame}

\begin{frame}{Mục tiêu học tập}
Sau bài học này, người học có thể:
\begin{enumerate}
  \item Sử dụng Pandas để kiểm tra cấu trúc, kiểu dữ liệu và giá trị thiếu.
  \item Phân biệt \textbf{missing value}, \textbf{outlier} và giá trị không hợp lệ.
  \item Sử dụng thống kê mô tả và biểu đồ (boxplot, histogram) để đánh giá dữ liệu.
  \item Áp dụng phương pháp IQR để phát hiện ngoại lệ và biết khi nào nên xử lý.
  \item Xử lý các biến phân loại bằng kỹ thuật \textbf{One-Hot Encoding}.
  \item Phân biệt và áp dụng đúng \textbf{Normalization} (Min-Max) và \textbf{Standardization} (Z-score).
  \item Hiểu nguyên tắc \textbf{chống Data Leakage}: Các phép biến đổi phải được học từ \textbf{Training set}.
  \item Xây dựng quy trình tiền xử lý hoàn chỉnh bằng \textbf{ColumnTransformer} và \textbf{Pipeline} của scikit-learn.
\end{enumerate}
\end{frame}

\section{Bước 5 — Phát hiện Ngoại lệ bằng Phương pháp IQR}

\begin{frame}{Công thức Interquartile Range (IQR)}
Phương pháp IQR dựa trên tứ phân vị, không bị ảnh hưởng bởi các giá trị cực trị:
\begin{itemize}
  \item $Q_1$ (Phân vị 25\%): 25\% dữ liệu nhỏ hơn giá trị này.
  \item $Q_3$ (Phân vị 75\%): 75\% dữ liệu nhỏ hơn giá trị này.
\end{itemize}

\begin{block}{Độ trải giữa (IQR) và Ngưỡng xác định}
\[
IQR = Q_3 - Q_1
\]
\[
\text{Ngưỡng dưới (Lower Bound)} = Q_1 - 1.5 \times IQR
\]
\[
\text{Ngưỡng trên (Upper Bound)} = Q_3 + 1.5 \times IQR
\]
\end{block}

\smallgap
Các giá trị nằm ngoài đoạn $[\text{Lower}, \text{Upper}]$ được coi là điểm ngoại lệ tiềm năng.
\end{frame}

\begin{frame}[fragile]{Cài đặt phương pháp IQR trong Python}
\begin{lstlisting}[style=pythonstyle]
# Tinh Q1, Q3 va IQR
q1, q3 = np.percentile(df["Insulin"].dropna(), [25, 75])
iqr = q3 - q1

lower = q1 - 1.5 * iqr
upper = q3 + 1.5 * iqr

print(f"Nguong duoi: {lower:.2f} | Nguong tren: {upper:.2f}")

# Loc ra cac quan sat ngoai le
outliers = df[(df["Insulin"] < lower) | (df["Insulin"] > upper)]
print("So luong ngoai le:", len(outliers))

# Neu co ly do hop ly de loai bo:
clean_df = df[(df["Insulin"] >= lower) & (df["Insulin"] <= upper)].copy()
\end{lstlisting}
\end{frame}

\begin{frame}{5 Câu hỏi trước khi xử lý Ngoại lệ}
Trước khi quyết định xử lý một điểm ngoại lệ, người phân tích cần tự hỏi:
\begin{enumerate}
  \item Giá trị này có \textbf{bất khả thi về mặt nghiệp vụ} không? (Ví dụ: Tuổi = -5 hay 200 tuổi).
  \item Có phải là \textbf{lỗi nhập liệu} hoặc lỗi hệ thống ghi nhận không? (Ví dụ: gõ thêm số 0).
  \item Có sự \textbf{sai lệch về đơn vị đo} không? (Ví dụ: gram vs.\ kilogram, VNĐ vs.\ USD).
  \item Ngoại lệ có phải chính là \textbf{đối tượng mục tiêu} cần phát hiện không? (Gian lận thẻ tín dụng, lỗi động cơ máy bay).
  \item Việc xóa có làm \textbf{mất đi một phân khúc quan trọng} của bài toán không?
\end{enumerate}

\smallgap
\textbf{Các phương án xử lý ngoài việc xóa bỏ:}
\begin{itemize}
  \item Giữ nguyên và dùng mô hình ít nhạy với outlier (Decision Tree, Random Forest).
  \item Cắt ngọn (Winsorization / Capping) đưa về ngưỡng $\text{Lower} / \text{Upper}$.
  \item Biến đổi dữ liệu (Log transform, Box-Cox) để thu hẹp độ phân tán.
\end{itemize}
\end{frame}

# Xoa cot neu thieu qua nhieu (>60%)
df_clean = df.drop(columns=["Col"])
\end{lstlisting}
\textit{Chỉ nên drop khi tỷ lệ thiếu rất nhỏ và không làm sai lệch phân phối.}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{2. Điền giá trị thống kê (Imputation):}
\begin{lstlisting}[style=pythonstyle]
# Dien bang Trung binh (Mean)
df["Age"] = df["Age"].fillna(
    df["Age"].mean()
)

# Dien bang Trung vi (Median) - tot khi co outlier
df["Income"] = df["Income"].fillna(
    df["Income"].median()
)

# Dien bien phan loai bang Mode
df["City"] = df["City"].fillna(
    df["City"].mode()[0]
)
\end{lstlisting}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{\texttt{SimpleImputer} trong Scikit-Learn}
Scikit-Learn cung cấp lớp \code{SimpleImputer} chuẩn hóa việc điền dữ liệu:
\begin{lstlisting}[style=pythonstyle]
from sklearn.impute import SimpleImputer

# Chien luoc dien trung vi cho bien so
imputer_num = SimpleImputer(strategy="median")

# Chien luoc dien yeu vi cho bien phan loai
imputer_cat = SimpleImputer(strategy="most_frequent")

# Chien luoc dien hang so
imputer_const = SimpleImputer(strategy="constant", fill_value="Unknown")
\end{lstlisting}

\smallgap
\textbf{Các tham số chính:}
\begin{itemize}
  \item \code{strategy}: \code{"mean"}, \code{"median"}, \code{"most\_frequent"}, \code{"constant"}.
  \item \code{fill\_value}: Giá trị cần điền khi dùng chiến lược \code{"constant"}.
\end{itemize}
\end{frame}

\begin{frame}{Cảnh báo quan trọng: Data Leakage khi Imputation}
\begin{alertblock}{Quy tắc phòng chống rò rỉ dữ liệu (Data Leakage)}
\textbf{KHÔNG BAO GIỜ} tính \code{mean} hay \code{median} trên toàn bộ tập dữ liệu rồi điền cho cả tập train và test!
\end{alertblock}

\smallgap
\begin{center}
\large
\[
\text{Dữ liệu Train} \longrightarrow \text{Học tham số (median)} \longrightarrow \text{Điền cho cả Train \& Test}
\]
\end{center}

\smallgap
\textbf{Lý do then chốt:}
\begin{itemize}
  \item Tập test đóng vai trò là dữ liệu mới trong tương lai.
  \item Nếu sử dụng thông tin của tập test để tính toán tham số tiền xử lý, mô hình sẽ đạt kết quả đánh giá quá lạc quan nhưng hoạt động kém khi triển khai thực tế.
\end{itemize}
\end{frame}

\section{Bước 7 — Xử lý Biến Phân loại (Categorical Encoding)}

\begin{frame}[fragile]{Vì sao cần mã hóa biến phân loại?}

\code{Region} \in \{\text{"North"}, \text{"Central"}, \text{"South"}\}
\]
cần được chuyển hóa thành các cột nhị phân (0 và 1).

\midrule
Central & 1 & 0 & 0 \\
North & 0 & 1 & 0 \\
South & 0 & 0 & 1 \\
\bottomrule
\end{tabularx}
\end{frame}

\begin{frame}[fragile]{\texttt{OneHotEncoder} trong Scikit-Learn}
Trong quy trình học máy chuyên nghiệp, nên dùng \code{OneHotEncoder} của scikit-learn:

\begin{lstlisting}[style=pythonstyle]
from sklearn.preprocessing import OneHotEncoder

\smallgap

Nếu tập Test xuất hiện một giá trị mới (ví dụ: \code{"Highlands"}) chưa từng có trong tập Train, \code{pd.get\_dummies} sẽ tạo số lượng cột khác nhau hoặc gây lỗi mô hình, còn \code{handle\_unknown="ignore"} sẽ gán toàn bộ số 0 cho các cột tương ứng một cách an toàn.
\end{block}
\end{frame}

  \item $y$ là nhãn cần dự báo.
  \item Đảm bảo trong $X$ không chứa thông tin của $y$ hoặc các biến chỉ xuất hiện sau khi $y$ đã xảy ra (\textbf{Target Leakage}).
\end{itemize}

\begin{frame}[fragile]{Bước 11: Chia Training Set và Test Set}

from sklearn.model_selection import train_test_split

X_train, X_test, y_train, y_test = train_test_split(
    X, y,
    test_size=0.2,       # 20% giu lai lam tap test
    random_state=42,     # Co dinh seed de tai lap ket qua
    stratify=y           # Giu nguyen ty le cac lop giua train va test
)
\end{lstlisting}

\smallgap
\textbf{Giải thích các tham số quan trọng:}
\begin{itemize}
  \item \code{test\_size=0.2}: 80\% dữ liệu dùng để học, 20\% dùng để kiểm định khách quan.
  \item \code{random\_state=42}: Đảm bảo mỗi lần chạy đều phân chia giống nhau.
  \item \code{stratify=y}: Đảm bảo tỷ lệ người mắc bệnh trong cả tập Train và Test đều giữ nguyên ở mức $\approx 35\%$.
\end{itemize}
\end{frame}

\section{Bước 12 — Chuẩn hóa Đặc trưng (Feature Scaling)}

\begin{frame}{Hai kỹ thuật chuẩn hóa đặc trưng cốt lõi}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{1. Min-Max Scaling (Normalization)}
Đưa dữ liệu về đoạn cố định, thường là $[0, 1]$:
\[
x' = \frac{x - x_{\min}}{x_{\max} - x_{\min}}
\]
\begin{itemize}
  \item Giá trị nhỏ nhất biến thành $0$.
  \item Giá trị lớn nhất biến thành $1$.
  \item Rất nhạy cảm với các điểm ngoại lệ (vì $x_{\min}$ và $x_{\max}$ bị kéo giãn).
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{2. Standardization (Z-score)}
Biến đổi dữ liệu có trung bình $0$ và độ lệch chuẩn $1$:
\[
z = \frac{x - \mu}{\sigma}
\]
\begin{itemize}
  \item Không giới hạn miền giá trị cố định.
  \item Ít bị chi phối bởi outlier hơn Min-Max.
  \item Không biến phân phối thành phân phối chuẩn nếu gốc không chuẩn!
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Thực hành Min-Max Scaling \& Kết quả}
\begin{lstlisting}[style=pythonstyle]

\small
\begin{tabularx}{\textwidth}{lXX}
\toprule
\textbf{Tiêu chí} & \textbf{Min-Max Scaling} & \textbf{Standardization} \\
\midrule
\textbf{Công thức} & $\frac{x - x_{\min}}{x_{\max} - x_{\min}}$ & $\frac{x - \mu}{\sigma}$ \\

\section{Bước 13 — Xây dựng Pipeline Tiền xử lý Chuyên nghiệp}

\begin{frame}[fragile]{Xử lý Dữ liệu Hỗn hợp với \texttt{ColumnTransformer}}
Dữ liệu thực tế thường gồm cả biến số và biến phân loại:
\begin{lstlisting}[style=tinypythonstyle]
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, OneHotEncoder

# Pipeline cho bien so: Dien median -> Chuan hoa Z-score
numeric_transformer = Pipeline(steps=[
    ("imputer", SimpleImputer(strategy="median")),
    ("scaler", StandardScaler())
])

# Pipeline cho bien phan loai: Dien mode -> One-Hot Encoding
categorical_transformer = Pipeline(steps=[
    ("imputer", SimpleImputer(strategy="most_frequent")),
    ("onehot", OneHotEncoder(handle_unknown="ignore"))
])

# Ket hop bang ColumnTransformer
preprocessor = ColumnTransformer(transformers=[
    ("num", numeric_transformer, ["Age", "Income", "Order_Value"]),
    ("cat", categorical_transformer, ["Region", "Customer_Type"])
])
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Tích hợp toàn bộ vào \texttt{Pipeline} Học máy}
Đóng gói toàn bộ Tiền xử lý và Mô hình dự báo:
\begin{lstlisting}[style=pythonstyle]
from sklearn.linear_model import LogisticRegression

# Dong goi preprocessor va model vao Pipeline chung
full_pipeline = Pipeline(steps=[
    ("preprocessor", preprocessor),
    ("classifier", LogisticRegression(max_iter=1000))
])

# Huan luyen: Scikit-learn tu dong fit tren train, khong leak du lieu!
full_pipeline.fit(X_train, y_train)

# Du doan tren tap test
y_pred = full_pipeline.predict(X_test)
\end{lstlisting}

\smallgap
\textbf{Lợi ích vượt trội của Pipeline:}
\begin{enumerate}
  \item Loại bỏ hoàn toàn nguy cơ rò rỉ dữ liệu (Data Leakage).
  \item Mã nguồn ngắn gọn, tường minh và dễ bảo trì.
  \item Triển khai dự báo trên dữ liệu mới (production) an toàn tuyệt đối.
\end{enumerate}
\end{frame}

\section{Quy trình Tổng quát \& Các Bẫy Sai sót Phổ biến}

\begin{frame}{Quy trình Tiền xử lý Dữ liệu Chuẩn mực}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Trình tự 10 bước an toàn:}
\begin{enumerate}
  \item \textbf{Understand:} Hiểu bài toán và ngữ cảnh.
  \item \textbf{Inspect:} Kiểm tra shape, info, dtypes.
  \item \textbf{Explore (EDA):} Thống kê mô tả, boxplot, heatmap.
  \item \textbf{Split:} Chia Train/Validation/Test.
  \item \textbf{Handle Missing:} Điền dữ liệu trên Train.
  \item \textbf{Encode:} Mã hóa biến phân loại.
  \item \textbf{Scale:} Chuẩn hóa thang đo biến số.
  \item \textbf{Pipeline:} Đóng gói các bước biến đổi.
  \item \textbf{Model:} Huấn luyện mô hình học máy.
  \item \textbf{Evaluate:} Đánh giá trên tập Test.
\end{enumerate}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Sơ đồ luồng an toàn}
\scriptsize
\begin{center}
Dữ liệu Thô \\
$\downarrow$ \\
\textbf{Train / Test Split} \\
$\downarrow$ \\
\textbf{Fit Preprocessing trên Train} \\
$\downarrow$ \\
Transform tập Train \\
$\downarrow$ \\
Huấn luyện Mô hình \\
$\downarrow$ \\
Transform tập Test theo tham số Train \\
$\downarrow$ \\
Đánh giá Mô hình Khách quan
\end{center}
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Những lỗi phổ biến khi Tiền xử lý (Phần 1: Dữ liệu \& Ngoại lệ)}
\begin{enumerate}
  \item \textbf{Tự động xóa mọi Outlier:}
  \begin{itemize}
    \item Sai vì ngoại lệ có thể là quan sát hoàn toàn hợp lệ hoặc là mục tiêu chính cần phát hiện (ví dụ: giao dịch gian lận).
  \end{itemize}
  \item \textbf{Chỉ kiểm tra \texttt{isna()}:}
  \begin{itemize}
    \item Bỏ sót các giá trị thiếu ngầm định ($0, -1, 999$, "?", "Unknown").
  \end{itemize}
  \item \textbf{Chuẩn hóa cả biến mục tiêu ($y$):}
  \begin{itemize}
    \item Với bài toán phân loại, $y$ là nhãn nhị phân ($0/1$), không được scale.
  \end{itemize}
  \item \textbf{Loại bỏ biến chỉ vì tương quan tuyến tính thấp:}
  \begin{itemize}
    \item Tương quan Pearson chỉ đo mối quan hệ tuyến tính; biến có thể chứa thông tin phi tuyến rất quan trọng.
  \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Những lỗi phổ biến khi Tiền xử lý (Phần 2: Leakage \& Đánh giá)}
\begin{enumerate}
  \setcounter{enumi}{4}
  \item \textbf{Fit Scaler / Imputer trước khi chia Train/Test:}
  \begin{itemize}
    \item Đây là dạng \textbf{Data Leakage} phổ biến nhất, khiến kết quả đánh giá mô hình bị sai lệch so với thực tế.
  \end{itemize}
  \item \textbf{Chọn biến có tương quan quá cao với $y$:}
  \begin{itemize}
    \item Nguy cơ \textbf{Target Leakage} (ví dụ biến \code{Sales\_Tax} được tính trực tiếp từ doanh số cần dự báo \code{Total\_Sales}).
  \end{itemize}
  \item \textbf{Xử lý tập Train và tập Test bằng các quy tắc khác nhau:}
  \begin{itemize}
    \item Gây sai lệch phân phối (distribution drift), mô hình dự đoán sai hoàn toàn khi triển khai trên dữ liệu mới.
  \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Tóm tắt Nội dung Bài học}
\begin{itemize}
  \item \textbf{Tiền xử lý dữ liệu} là nền tảng quyết định chất lượng của mọi mô hình học máy.
  \item \textbf{Missing Values:} Luôn lưu ý các giá trị thiếu tiềm ẩn ($0, -1, 999$); dùng \code{SimpleImputer}.
  \item \textbf{Outliers:} Dùng Boxplot và phương pháp IQR để phát hiện; tìm hiểu nguyên nhân trước khi quyết định xử lý.
  \item \textbf{Categorical Encoding:} Sử dụng One-Hot Encoding với \code{handle\_unknown="ignore"} để xử lý nhãn mới.
  \item \textbf{Scaling:} Phân biệt Min-Max Scaling (đoạn $[0,1]$) và Standardization ($\mu=0, \sigma=1$).
  \item \textbf{Chống Data Leakage:} Luôn chia Train/Test trước; chỉ \code{fit} trên Train, dùng tham số đó để \code{transform} Test.
  \item \textbf{Pipeline \& ColumnTransformer:} Công cụ chuẩn mực của scikit-learn giúp đóng gói toàn bộ quy trình tiền xử lý an toàn và chuyên nghiệp.
\end{itemize}
\end{frame}
