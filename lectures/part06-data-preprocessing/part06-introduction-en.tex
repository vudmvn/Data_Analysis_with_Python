\section{Overview of Data Preprocessing}

\begin{frame}{Motivation}
In practice, raw data is rarely ready to be fed directly into machine learning models:
\begin{itemize}
  \item \textbf{Missing values:} caused by user omissions or data transmission errors.
  \item \textbf{Outliers:} could be clerical errors or rare, high-value observations.
  \item \textbf{Improper data types:} numeric variables stored as strings, incorrect datetime formats.
  \item \textbf{Categorical variables:} text labels that need to be transformed into numerical formats.
  \item \textbf{Scale discrepancies:} income in millions vs.\ age under 100.
  \item \textbf{Class imbalance:} fraud detection or medical diagnoses with low positive rates.
\end{itemize}

\smallgap
\begin{alertblock}{Core Principle}
\textbf{Garbage In, Garbage Out:} Complex machine learning algorithms trained on dirty data produce unreliable and biased results.
\end{alertblock}
\end{frame}

\begin{frame}{Position in the Machine Learning Workflow}

\begin{column}{0.50\textwidth}
\textbf{Key Benefits:}
\begin{itemize}
  \item Improves the quality of model inputs.
  \item Mitigates the harmful impact of noise.
  \item Ensures distance-based and gradient-based algorithms work correctly.
  \item \textbf{Prevents Data Leakage:} Eliminates leakage of test information into the training phase.
  \item Makes model decisions more interpretable.
\end{itemize}
\end{column}

\end{frame}

\begin{frame}{Why Do Algorithms Need Preprocessing?}
Many machine learning algorithms are sensitive to feature scales:
\begin{itemize}
  \item Consider two features: \code{Income} (5,000,000 to 100,000,000) and \code{Age} (18 to 70).
  \item Using Euclidean distance:
  \[
  d = \sqrt{(\text{Income}_1 - \text{Income}_2)^2 + (\text{Age}_1 - \text{Age}_2)^2}
  \]
  \item \code{Income} completely \textbf{dominates} the distance calculation, rendering \code{Age} irrelevant!
\end{itemize}

\smallgap
\textbf{Heavily Impacted Algorithms:}
\begin{itemize}
  \item k-Nearest Neighbors (k-NN), k-Means Clustering
  \item Support Vector Machines (SVM)
  \item Regularized Regression (Ridge, Lasso, ElasticNet)
  \item Artificial Neural Networks (Deep Learning)
\end{itemize}
\end{frame}

\begin{frame}{Understanding Outliers}
An \textbf{outlier} is an observation that lies an abnormal distance from other values in the sample.

\smallgap
\begin{alertblock}{Core Reality}
\centering
\textbf{Outliers are NOT inherently erroneous data!}
\end{alertblock}

\smallgap
\textbf{Practical Examples:}
\begin{itemize}
  \item A VIP corporate client spending \$50,000 compared to an average of \$200.
  \item An international delivery shipment taking 45 days compared to a 3-day baseline.
  \item In \textbf{Fraud Detection}, rare fraudulent transactions are precisely the outliers we aim to detect!
\end{itemize}

\textbf{Guiding Rule:} The first priority is to \textbf{investigate the context and cause}, never delete outliers blindly.
\end{frame}

\section{Step 5 — Outlier Detection using the IQR Method}

\begin{frame}{Interquartile Range (IQR) Formula}
The IQR method relies on quartiles, making it robust against extreme values:
\begin{itemize}
  \item $Q_1$ (25th Percentile): 25\% of the data falls below this value.
  \item $Q_3$ (75th Percentile): 75\% of the data falls below this value.
\end{itemize}

\begin{block}{Interquartile Range and Detection Thresholds}
\[
IQR = Q_3 - Q_1
\]
\[
\text{Lower Bound} = Q_1 - 1.5 \times IQR
\]
\[
\text{Upper Bound} = Q_3 + 1.5 \times IQR
\]
\end{block}

\smallgap
Observations falling outside $[\text{Lower Bound}, \text{Upper Bound}]$ are flagged as candidate outliers.
\end{frame}

\begin{frame}[fragile]{Implementing the IQR Method in Python}
\begin{lstlisting}[style=pythonstyle]
# Compute Q1, Q3, and IQR
q1, q3 = np.percentile(df["Insulin"].dropna(), [25, 75])
iqr = q3 - q1

lower = q1 - 1.5 * iqr
upper = q3 + 1.5 * iqr

print(f"Lower Bound: {lower:.2f} | Upper Bound: {upper:.2f}")

# Filter outlier records
outliers = df[(df["Insulin"] < lower) | (df["Insulin"] > upper)]
print("Total Outliers Detected:", len(outliers))

# If justifiable to filter:
clean_df = df[(df["Insulin"] >= lower) & (df["Insulin"] <= upper)].copy()
\end{lstlisting}
\end{frame}

\begin{frame}{5 Questions to Ask Before Removing Outliers}
Before discarding any outlier, ask:
\begin{enumerate}
  \item Is the value \textbf{physically impossible} in the given business domain? (e.g., Age = -5 or 250).
  \item Is it a known \textbf{data entry error} or logging glitch? (e.g., typing an extra zero).
  \item Is there a \textbf{unit mismatch}? (e.g., grams vs.\ kilograms, cents vs.\ dollars).
  \item Is the outlier the \textbf{primary object of interest}? (e.g., fraud, equipment failure).
  \item Does removing it discard an \textbf{important customer segment}?
\end{enumerate}

\smallgap
\textbf{Alternative Treatment Strategies:}
\begin{itemize}
  \item Retain the data and adopt robust models (Decision Trees, Random Forests).
  \item Winsorization / Capping (clipping values to Lower/Upper bounds).
  \item Mathematical transformations (Log, Box-Cox) to compress variance.
\end{itemize}
\end{frame}

\section{Step 6 — Handling Missing Values}

\begin{frame}[fragile]{Basic Missing Data Strategies}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{1. Deletion (Rows or Columns):}
\begin{lstlisting}[style=pythonstyle]
# Drop all rows containing missing values
df_clean = df.dropna()

# Drop rows missing specific features
df_clean = df.dropna(subset=["Age"])

# Drop columns missing > 60% of entries
df_clean = df.drop(columns=["Col"])
\end{lstlisting}
\textit{Only delete when missingness is negligible and random.}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{2. Statistical Imputation:}
\begin{lstlisting}[style=pythonstyle]
# Impute with Mean
df["Age"] = df["Age"].fillna(
    df["Age"].mean()
)

# Impute with Median (robust to outliers)
df["Income"] = df["Income"].fillna(
    df["Income"].median()
)

# Impute Categoricals with Mode
df["City"] = df["City"].fillna(
    df["City"].mode()[0]
)
\end{lstlisting}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{\texttt{SimpleImputer} in Scikit-Learn}
Scikit-Learn provides \code{SimpleImputer} for standardized, production-ready imputation:
\begin{lstlisting}[style=pythonstyle]
from sklearn.impute import SimpleImputer

# Median imputation for numerical features
imputer_num = SimpleImputer(strategy="median")

# Mode imputation for categorical features
imputer_cat = SimpleImputer(strategy="most_frequent")

# Constant value imputation
imputer_const = SimpleImputer(strategy="constant", fill_value="Unknown")
\end{lstlisting}

\smallgap
\textbf{Key Parameters:}
\begin{itemize}
  \item \code{strategy}: \code{"mean"}, \code{"median"}, \code{"most\_frequent"}, \code{"constant"}.
  \item \code{fill\_value}: Replacement value when \code{strategy="constant"}.
\end{itemize}
\end{frame}

\begin{frame}{Critical Warning: Data Leakage During Imputation}
\begin{alertblock}{Data Leakage Prevention Principle}
\textbf{NEVER} compute statistics (\code{mean}, \code{median}) across the entire dataset before splitting into train and test sets!
\end{alertblock}

\smallgap
\begin{center}
\large
\[
\text{Training Data} \longrightarrow \text{Fit Parameter (e.g., Median)} \longrightarrow \text{Transform Both Train \& Test}
\]
\end{center}

\smallgap
\textbf{Why this is mandatory:}
\begin{itemize}
  \item The test set must simulate unseen future production data.
  \item Computing imputation values using the test set introduces information leakage, yielding overoptimistic test metrics that degrade in deployment.
\end{itemize}
\end{frame}

\section{Step 7 — Handling Categorical Variables (Encoding)}

\begin{frame}[fragile]{Why Is Categorical Encoding Necessary?}
Most machine learning algorithms (Linear/Logistic Regression, Neural Networks, SVM) operate strictly on numerical matrices. Hence, qualitative labels:
\[
\code{Region} \in \{\text{"North"}, \text{"Central"}, \text{"South"}\}
\]
must be converted into numerical binary representations.

\smallgap
\textbf{Method 1: One-Hot Encoding via Pandas (\code{pd.get\_dummies})}
\begin{lstlisting}[style=pythonstyle]
# drop_first=True prevents multicollinearity (dummy variable trap)
df_encoded = pd.get_dummies(df, columns=["Region"], drop_first=True)
\end{lstlisting}

\smallgap
\begin{tabularx}{\textwidth}{lXXX}
\toprule
\textbf{Region (Original)} & \textbf{Region\_Central} & \textbf{Region\_North} & \textbf{Region\_South} (Dropped if \code{drop\_first}) \\
\midrule
Central & 1 & 0 & 0 \\
North & 0 & 1 & 0 \\
South & 0 & 0 & 1 \\
\bottomrule
\end{tabularx}
\end{frame}

\begin{frame}[fragile]{\texttt{OneHotEncoder} in Scikit-Learn}
For robust production pipelines, Scikit-Learn's \code{OneHotEncoder} is the gold standard:

\begin{lstlisting}[style=pythonstyle]
from sklearn.preprocessing import OneHotEncoder

# handle_unknown='ignore' is crucial for unseen test labels
encoder = OneHotEncoder(handle_unknown="ignore", sparse_output=False)

# Learn categories from training set and transform
X_train_encoded = encoder.fit_transform(X_train[["Region"]])
X_test_encoded = encoder.transform(X_test[["Region"]])
\end{lstlisting}

\smallgap
\begin{block}{Why is \texttt{handle\_unknown="ignore"} Essential?}
If an unseen category appears during testing (e.g., \code{"Highlands"}), \code{pd.get\_dummies} would yield inconsistent column structures or crash, whereas \code{handle\_unknown="ignore"} encodes all category columns with zeros safely.
\end{block}
\end{frame}

\section{Steps 10 \& 11 — Feature Splitting and Train/Test Split}

\begin{frame}[fragile]{Step 10: Splitting Predictors ($X$) and Target ($y$)}
\begin{lstlisting}[style=pythonstyle]
# X contains input predictors (features)
X = df.drop(columns=["Outcome"])

# y contains the target label to predict
y = df["Outcome"]

print("Predictor Matrix Shape (X):", X.shape)
print("Target Vector Shape (y):", y.shape)
\end{lstlisting}

\smallgap
\textbf{Significance:}
\begin{itemize}
  \item $X$ contains information available at prediction time.
  \item $y$ represents the ground-truth outcome.
  \item Ensure $X$ never includes variables calculated after the target occurs (\textbf{Target Leakage}).
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Step 11: Train/Test Data Partitioning}
\begin{lstlisting}[style=pythonstyle]
from sklearn.model_selection import train_test_split

X_train, X_test, y_train, y_test = train_test_split(
    X, y,
    test_size=0.2,       # 20% reserved for holdout evaluation
    random_state=42,     # Reproducibility seed
    stratify=y           # Preserves class proportions across splits
)
\end{lstlisting}

\smallgap
\textbf{Key Argument Explanations:}
\begin{itemize}
  \item \code{test\_size=0.2}: 80\% used for model fitting, 20\% reserved for objective testing.
  \item \code{random\_state=42}: Guarantees identical data partitions across runs.
  \item \code{stratify=y}: Maintains identical target class ratios ($\approx 35\%$) across both train and test splits.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{The Golden Rule: Why Split Before Scaling?}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{alertblock}{WRONG (Data Leakage)}
\begin{lstlisting}[style=pythonstyle]
scaler = StandardScaler()
# Scaler fits on ENTIRE dataset
X_scaled = scaler.fit_transform(X)

# Split performed post-scaling
X_tr, X_te = train_test_split(X_scaled)
\end{lstlisting}
$\rightarrow$ Test set mean $\mu$ and variance $\sigma^2$ leak into the model training process!
\end{alertblock}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{CORRECT PRACTICE}
\begin{lstlisting}[style=pythonstyle]
# Partition train/test first
X_tr, X_te, y_tr, y_te = (
    train_test_split(X, y, test_size=0.2)
)

scaler = StandardScaler()
# ONLY fit on Train
X_tr_scaled = scaler.fit_transform(X_tr)
# Test is ONLY transformed
X_te_scaled = scaler.transform(X_te)
\end{lstlisting}
\end{block}
\end{column}
\end{columns}
\end{frame}

\section{Step 12 — Feature Scaling (Min-Max vs. Standardization)}

\begin{frame}{Two Foundational Scaling Paradigms}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{1. Min-Max Scaling (Normalization)}
Maps feature values into a bounded interval, typically $[0, 1]$:
\[
x' = \frac{x - x_{\min}}{x_{\max} - x_{\min}}
\]
\begin{itemize}
  \item Minimum value transforms to $0$.
  \item Maximum value transforms to $1$.
  \item Sensitive to extreme outliers because $x_{\min}$ and $x_{\max}$ anchor the scale.
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{2. Standardization (Z-score)}
Centers data around mean $0$ with standard deviation $1$:
\[
z = \frac{x - \mu}{\sigma}
\]
\begin{itemize}
  \item Does not constrain data to bounded limits.
  \item Less vulnerable to extreme outliers than Min-Max.
  \item Does not transform non-normal distributions into Gaussian shapes!
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Min-Max Scaling Implementation \& Output}
\begin{lstlisting}[style=pythonstyle]
from sklearn.preprocessing import MinMaxScaler

minmax_scaler = MinMaxScaler()

X_test_norm = minmax_scaler.transform(X_test)
\end{lstlisting}

\begin{frame}{Min-Max Scaling vs. Standardization}
\small
\begin{tabularx}{\textwidth}{lXX}
\toprule
\textbf{Evaluation Dimension} & \textbf{Min-Max Scaling} & \textbf{Standardization} \\
\midrule
\textbf{Mathematical Formula} & $\frac{x - x_{\min}}{x_{\max} - x_{\min}}$ & $\frac{x - \mu}{\sigma}$ \\
\textbf{Output Range} & Typically bounded $[0, 1]$ & Unbounded $(-\infty, +\infty)$ \\
\textbf{Resulting Mean} & Not necessarily $0$ & Equal to $0$ on training set \\
\textbf{Resulting Standard Dev.} & Variable & Equal to $1$ on training set \\
\textbf{Sensitivity to Outliers} & \textbf{Very High} & Moderate \\
\textbf{Typical Applications} & Bounded algorithms, image processing (0--255 pixels), k-NN & Ridge/Lasso regression, SVM, PCA, Neural Networks \\
\bottomrule
\end{tabularx}
\end{frame}

\begin{frame}{Which Algorithms Require Feature Scaling?}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{REQUIRES Feature Scaling}
Models relying on distance metrics or gradient descent optimization:
\begin{itemize}
  \item k-Nearest Neighbors (k-NN)
  \item k-Means Clustering
  \item Support Vector Machines (SVM)
  \item Principal Component Analysis (PCA)
  \item Regularized Linear/Logistic Models
  \item Deep Neural Networks
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{DOES NOT Require Scaling}
Tree-based partition algorithms:
\begin{itemize}
  \item Decision Trees
  \item Random Forests
  \item Gradient Boosted Trees (XGBoost, LightGBM, CatBoost)
\end{itemize}
\smallgap
\textit{Rationale:} Trees partition features monotonically and independently; splits are invariant to linear scale shifts.
\end{block}
\end{column}
\end{columns}
\end{frame}

\section{Step 13 — Building Preprocessing Pipelines with Scikit-Learn}

\begin{frame}[fragile]{Handling Mixed Datasets with \texttt{ColumnTransformer}}
Real-world datasets contain both continuous numerical and categorical columns:
\begin{lstlisting}[style=tinypythonstyle]
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, OneHotEncoder

# Pipeline for numerical features: Median Imputation -> Standardization
numeric_transformer = Pipeline(steps=[
    ("imputer", SimpleImputer(strategy="median")),
    ("scaler", StandardScaler())
])

# Pipeline for categorical features: Mode Imputation -> One-Hot Encoding
categorical_transformer = Pipeline(steps=[
    ("imputer", SimpleImputer(strategy="most_frequent")),
    ("onehot", OneHotEncoder(handle_unknown="ignore"))
])

# Assemble combined preprocessor via ColumnTransformer
preprocessor = ColumnTransformer(transformers=[
    ("num", numeric_transformer, ["Age", "Income", "Order_Value"]),
    ("cat", categorical_transformer, ["Region", "Customer_Type"])
])
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Full Model Integration via \texttt{Pipeline}}
Encapsulate preprocessing and estimator into a unified object:
\begin{lstlisting}[style=pythonstyle]
from sklearn.linear_model import LogisticRegression

# Bundle preprocessor and model estimator together
full_pipeline = Pipeline(steps=[
    ("preprocessor", preprocessor),
    ("classifier", LogisticRegression(max_iter=1000))
])

# Fit pipeline: Learned strictly from train set, zero leakage risk!
full_pipeline.fit(X_train, y_train)

# Inference on unseen test data
y_pred = full_pipeline.predict(X_test)
\end{lstlisting}

\smallgap
\textbf{Major Pipeline Advantages:}
\begin{enumerate}
  \item Completely eliminates data leakage across validation folds.
  \item Concise, readable, and maintainable codebase.
  \item Seamless transition from development to production inference.
\end{enumerate}
\end{frame}

\section{General Workflow and Common Pitfalls}

\begin{frame}{Standard End-to-End Preprocessing Workflow}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{10-Step Robust Sequence:}
\begin{enumerate}
  \item \textbf{Understand:} Business domain and features.
  \item \textbf{Inspect:} Examine shape, info, types.
  \item \textbf{Explore (EDA):} Distributions, boxplots, correlations.
  \item \textbf{Split:} Partition Train/Validation/Test sets.
  \item \textbf{Impute:} Treat missing values from Train.
  \item \textbf{Encode:} Transform categorical columns.
  \item \textbf{Scale:} Standardize or normalize features.
  \item \textbf{Pipeline:} Wrap transformations cleanly.
  \item \textbf{Model:} Fit predictive estimators.
  \item \textbf{Evaluate:} Assess performance on Test set.
\end{enumerate}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Execution Flowchart}
\scriptsize
\begin{center}
Raw Data \\
$\downarrow$ \\
\textbf{Train / Test Partition} \\
$\downarrow$ \\
\textbf{Fit Preprocessing on Train Only} \\
$\downarrow$ \\
Transform Training Set \\
$\downarrow$ \\
Train ML Estimator \\
$\downarrow$ \\
Transform Test Set with Learned Parameters \\
$\downarrow$ \\
Evaluate Real-World Performance
\end{center}
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Common Preprocessing Traps (Part 1: Data \& Outliers)}
\begin{enumerate}
  \item \textbf{Automated Deletion of All Outliers:}
  \begin{itemize}
    \item Flawed because outliers often represent valid high-value entities or crucial target signals (e.g., financial fraud).
  \end{itemize}
  \item \textbf{Solely Relying on \texttt{isna()}:}
  \begin{itemize}
    \item Misses implicit sentinel representations ($0, -1, 999$, "?", "Unknown").
  \end{itemize}
  \item \textbf{Scaling Target Labels ($y$):}
  \begin{itemize}
    \item In classification, target labels are discrete categorical classes ($0/1$) and must never be scaled.
  \end{itemize}
  \item \textbf{Discarding Features Due to Low Linear Correlation:}
  \begin{itemize}
    \item Pearson correlation only reflects linear relationships; valuable non-linear patterns may be lost.
  \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Common Preprocessing Traps (Part 2: Leakage \& Evaluation)}
\begin{enumerate}
  \setcounter{enumi}{4}
  \item \textbf{Fitting Scalers / Imputers Prior to Train/Test Split:}
  \begin{itemize}
    \item The most pervasive form of \textbf{Data Leakage}, distorting performance estimates.
  \end{itemize}
  \item \textbf{Selecting Features with Overly High Target Correlation:}
  \begin{itemize}
    \item Danger of \textbf{Target Leakage} (e.g., using \code{Sales\_Tax} calculated directly from the forecast target \code{Total\_Sales}).
  \end{itemize}
  \item \textbf{Inconsistent Processing of Train vs.\ Test Sets:}
  \begin{itemize}
    \item Induces distribution drift, leading to catastrophic inference failures in production.
  \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Case Study: Customer Churn Prediction}
\textbf{Telecommunications Dataset Features:}
\begin{itemize}
  \item \code{Customer\_ID, Age, Monthly\_Spending, Tenure\_Months, Region, Support\_Calls, Churn}
\end{itemize}

\textbf{Standard Preprocessing Steps:}
\begin{enumerate}
  \item Drop \code{Customer\_ID} (arbitrary identifier with zero predictive power).
  \item Detect \code{Age = -1} $\rightarrow$ convert to \code{NaN}, then impute with median.
  \item Clean \code{Region}: Harmonize \code{"Hanoi", "Ha Noi", "HN"} into \code{"Hanoi"}.
  \item Evaluate \code{Support\_Calls} outliers via boxplots; retain them as high call volume strongly signals churn risk.
  \item Separate $X, y$ and partition with \code{stratify=y}.
  \item Build Pipeline: Impute \& scale numerical variables; One-hot encode categoricals.
\end{enumerate}
\end{frame}

\begin{frame}{Lecture Summary}
\begin{itemize}
  \item \textbf{Data Preprocessing} forms the critical bedrock of all predictive modeling.
  \item \textbf{Missing Values:} Beware of implicit sentinels ($0, -1, 999$); standardize using \code{SimpleImputer}.
  \item \textbf{Outliers:} Diagnose using Boxplots and the IQR method; inspect context before deciding to filter or winsorize.
  \item \textbf{Categorical Encoding:} Utilize One-Hot Encoding with \code{handle\_unknown="ignore"} to ensure inference safety.
  \item \textbf{Feature Scaling:} Differentiate Min-Max Scaling ($[0,1]$ bounds) and Standardization ($\mu=0, \sigma=1$).
  \item \textbf{Leakage Prevention:} Always partition Train/Test first; only \code{fit} on Train and \code{transform} Test.
  \item \textbf{Pipelines \& ColumnTransformer:} Scikit-Learn's core tools to build robust, modular, and leak-free workflows.
\end{itemize}
\end{frame}
