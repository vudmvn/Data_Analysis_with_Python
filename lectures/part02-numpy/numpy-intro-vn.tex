\section{NumPy là gì?}

\begin{frame}{NumPy là gì?}
\small
\key{NumPy}, viết tắt của \key{Numerical Python}, là thư viện Python cốt lõi cho tính toán số nhanh và hiệu quả.
\vspace{0.4em}

Đối tượng trung tâm của NumPy là \key{mảng N chiều}, gọi là \texttt{ndarray}. Một mảng NumPy có thể biểu diễn:
\begin{itemize}
  \item Vector một chiều.
  \item Ma trận hai chiều.
  \item Tensor ba chiều.
  \item Các cấu trúc số nhiều chiều hơn.
\end{itemize}
\begin{block}{Ý tưởng chính}
NumPy lưu trữ dữ liệu đồng nhất hiệu quả và hỗ trợ phép toán nhanh trên toàn bộ mảng.
\end{block}
\end{frame}

\begin{frame}{NumPy cung cấp gì?}
\small
\begin{columns}[T]
\column{0.47\linewidth}
\begin{itemize}
  \item Phép toán nhanh trên mảng.
  \item Tính toán vector hóa.
  \item Broadcasting.
  \item Hàm đại số tuyến tính.
\end{itemize}
\column{0.47\linewidth}
\begin{itemize}
  \item Hàm thống kê.
  \item Sinh số ngẫu nhiên.
  \item Tích hợp với Pandas, SciPy, Matplotlib và Scikit-learn.
  \item Nền tảng cho nhiều quy trình khoa học dữ liệu.
\end{itemize}
\end{columns}
% \imageplaceholder{numpy_linear_algebra.webp}{Tổng quan NumPy và đại số tuyến tính}
\end{frame}

\begin{frame}{Các đặc trưng chính}
\small
\begin{block}{\texttt{ndarray}}
Cấu trúc dữ liệu trung tâm, lưu trữ phần tử cùng kiểu trong mảng nhiều chiều.
\end{block}
\begin{block}{Vector hóa và broadcasting}
Vector hóa áp dụng phép tính lên toàn bộ mảng; broadcasting cho phép thao tác giữa các hình dạng tương thích.
\end{block}
\begin{block}{Đại số tuyến tính và thống kê}
NumPy hỗ trợ nhân ma trận, định thức, nghịch đảo, trị riêng, trung bình, trung vị, phương sai và phân vị.
\end{block}
\end{frame}

\section{Vì sao nên học NumPy?}

\begin{frame}{Vì sao nên học NumPy?}
\small
NumPy là nền tảng hiệu quả cho tính toán số trong Python.
\begin{itemize}
  \item Nhanh hơn nhiều vòng lặp Python trong nhiều tác vụ số nhờ vector hóa.
  \item Lưu trữ dữ liệu đồng nhất gọn hơn list Python.
  \item Cung cấp hàm tối ưu cho ma trận, đại số tuyến tính và thống kê.
  \item Cho phép biểu diễn công thức phức tạp bằng lệnh ngắn gọn.
  \item Là nền tảng cho Pandas, SciPy, Scikit-learn và nhiều thư viện dữ liệu khác.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{List Python và mảng NumPy}
\small
\begin{columns}[T]
\column{0.48\linewidth}
\begin{block}{List Python}
\begin{lstlisting}[style=py]
values = [10, 2.5, "Python", True]
\end{lstlisting}
\begin{itemize}
  \item Có thể chứa nhiều kiểu dữ liệu.
  \item Thao tác số thường cần vòng lặp.
\end{itemize}
\end{block}
\column{0.48\linewidth}
\begin{block}{Mảng NumPy}
\begin{lstlisting}[style=py]
import numpy as np
values = np.array([10, 20, 30, 40])
\end{lstlisting}
\begin{itemize}
  \item Thường chứa một kiểu dữ liệu chung.
  \item Hỗ trợ phép toán số vector hóa.
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Ví dụ: nhân toàn bộ giá trị}
\small
\begin{columns}[T]
\column{0.48\linewidth}
\begin{block}{Dùng list Python}
\begin{lstlisting}[style=py]
values = [1, 2, 3, 4]
result = []
for value in values:
    result.append(value * 10)
print(result)
\end{lstlisting}
\end{block}
\column{0.48\linewidth}
\begin{block}{Dùng NumPy}
\begin{lstlisting}[style=py]
import numpy as np
values = np.array([1, 2, 3, 4])
result = values * 10
print(result)
\end{lstlisting}
\end{block}
\end{columns}
\vspace{0.4em}
\begin{alertblock}{Điểm cần nhớ}
Trong NumPy, \texttt{values * 10} được áp dụng cho toàn bộ mảng mà không cần vòng lặp tường minh.
\end{alertblock}
\end{frame}

\section{Tạo mảng NumPy}

\begin{frame}[fragile]{Tạo mảng NumPy từ list}
\small
\begin{lstlisting}[style=py]
import numpy as np

a = [9, 3, 3, 5]
arr = np.array(a)
print(arr)
\end{lstlisting}
\begin{block}{Kết quả}
\texttt{[9 3 3 5]}
\end{block}
\begin{itemize}
  \item \texttt{np.array()} chuyển list Python thành mảng NumPy.
  \item Mảng một chiều thường được dùng để biểu diễn vector.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Mảng nhiều chiều}
\small
\begin{columns}[T]
\column{0.48\linewidth}
\begin{block}{Mảng hai chiều}
\begin{lstlisting}[style=py]
matrix = np.array([
    [1, 2, 3],
    [4, 5, 6]
])
print(matrix)
\end{lstlisting}
\end{block}
\column{0.48\linewidth}
\begin{block}{Mảng ba chiều}
\begin{lstlisting}[style=py]
tensor = np.array([
    [[1, 2], [3, 4]],
    [[5, 6], [7, 8]]
])
print(tensor)
\end{lstlisting}
\end{block}
\end{columns}
\begin{itemize}
  \item Mảng hai chiều thường là ma trận.
  \item Mảng ba chiều có thể xem như tập nhiều ma trận hoặc tensor.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Các hàm tạo mảng thông dụng}
\small

\end{lstlisting}
\end{columns}
\vspace{0.5em}
\begin{block}{Ý nghĩa}
\begin{itemize}
  \item \texttt{arange}: tạo dãy theo bước nhảy.
  \item \texttt{linspace}: tạo số lượng điểm cách đều giữa hai đầu mút.
  \item \texttt{eye}: tạo ma trận đơn vị.

\end{block}
\end{frame}

\begin{frame}[fragile]{Slicing trong NumPy}
\small

\begin{frame}[fragile]{View và Copy}

\end{block}

\begin{lstlisting}[style=py]

\end{lstlisting}
\end{frame}

\section{Broadcasting và phép toán}

\begin{frame}[fragile]{Broadcasting là gì?}
\small
Broadcasting cho phép NumPy thực hiện phép toán giữa các mảng có hình dạng khác nhau nhưng tương thích.
\begin{block}{Broadcasting với số vô hướng}
\begin{lstlisting}[style=py]
arr = np.array([1, 2, 3, 4])
result = arr + 10
print(result)
\end{lstlisting}
\texttt{[11 12 13 14]}
\end{block}
\begin{block}{Quy tắc cơ bản}
So sánh từ chiều ngoài cùng bên phải. Hai chiều tương thích nếu chúng bằng nhau hoặc một trong hai bằng \texttt{1}.
\end{block}
\end{frame}

\begin{frame}[fragile]{Broadcasting giữa các mảng}
\small
\begin{lstlisting}[style=py]
matrix = np.array([
    [1, 2, 3],
    [4, 5, 6]
])
row = np.array([10, 20, 30])
result = matrix + row
print(result)
\end{lstlisting}
\begin{block}{Kết quả}
\texttt{[[11 22 33]\ [14 25 36]]}
\end{block}

\section{Ufuncs, Boolean, Linear Algebra}

\begin{frame}[fragile]{Universal Functions}
\small
\key{Universal function} hay \key{ufunc} thực hiện phép toán theo từng phần tử trên mảng.
\begin{columns}[T]
\column{0.48\linewidth}
\begin{lstlisting}[style=py]
arr = np.array([1, 4, 9, 16])
print(np.sqrt(arr))
print(np.exp(np.array([0, 1, 2])))
\end{lstlisting}
\column{0.48\linewidth}
\begin{lstlisting}[style=py]
values = np.array([-3, -1, 2, 4])
print(np.abs(values))
print(np.round([1.234, 5.678], 2))
\end{lstlisting}
\end{columns}
\begin{block}{Ví dụ khác}
\texttt{np.log()}, \texttt{np.sin()}, \texttt{np.cos()}, \texttt{np.tan()}.
\end{block}
\end{frame}

\begin{frame}[fragile]{So sánh và lọc Boolean}
\small
\begin{lstlisting}[style=py]
arr = np.array([10, 20, 30, 40])
print(arr > 20)
selected = arr[arr > 20]
print(selected)
\end{lstlisting}
\begin{lstlisting}[style=py]
selected = arr[(arr >= 20) & (arr <= 30)]
print(selected)
\end{lstlisting}
\begin{itemize}
  \item \texttt{\&}: AND theo từng phần tử.
  \item \texttt{|}: OR theo từng phần tử.
  \item \texttt{\textasciitilde}: NOT theo từng phần tử.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Đại số tuyến tính trong NumPy}
\small
\begin{lstlisting}[style=py]
A = np.array([
    [1, 2],
    [3, 4]
])

print(np.dot(A, A))
print(A @ A)
print(A.T)
print(np.linalg.det(A))
print(np.linalg.inv(A))
\end{lstlisting}
\begin{block}{Ghi nhớ}
\texttt{A * A} là nhân theo từng phần tử; \texttt{A @ A} là nhân ma trận.
\end{block}
\end{frame}

\section{Lỗi thường gặp và thực hành tốt}

\begin{frame}[fragile]{Các lỗi NumPy thường gặp}
\small
\begin{block}{Không khớp hình dạng}
\begin{lstlisting}[style=py]
a = np.ones((2, 3))
b = np.ones((2, 2))
# a + b raises an error
\end{lstlisting}
\end{block}
\begin{block}{Reshape không hợp lệ}
\begin{lstlisting}[style=py]
arr = np.arange(10)
# arr.reshape(3, 4) raises an error
\end{lstlisting}
\end{block}
\begin{itemize}
  \item Chia cho 0 có thể tạo \texttt{inf}.
  \item Kiểu số nguyên nhỏ có thể bị tràn.
  \item View có thể làm thay đổi mảng gốc ngoài ý muốn.
\end{itemize}
\end{frame}

\begin{frame}{Thực hành tốt khi dùng NumPy}
\small
\begin{itemize}
  \item Import NumPy bằng \texttt{import numpy as np}.
  \item Ưu tiên vector hóa thay vì vòng lặp khi phù hợp.
  \item Kiểm tra \texttt{shape}, \texttt{ndim}, và \texttt{dtype} trước phép toán phức tạp.
  \item Chỉ dùng broadcasting khi hiểu rõ hành vi hình dạng.
  \item Phân biệt \texttt{*} và \texttt{@}.
  \item Dùng \texttt{.copy()} khi cần mảng độc lập.
  \item Chọn \texttt{dtype} phù hợp với phạm vi, độ chính xác và bộ nhớ.
  \item Dùng seed cố định khi cần tái lập kết quả.
  \item Xử lý rõ \texttt{NaN}, \texttt{inf} và tràn số.
\end{itemize}
\end{frame}
