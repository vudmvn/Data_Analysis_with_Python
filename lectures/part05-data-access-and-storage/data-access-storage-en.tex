\begin{frame}[fragile]{Reading CSV Data with NumPy}
\small
NumPy provides two primary functions for text-based matrix loading:
\begin{columns}[T]
\column{0.48\linewidth}
\begin{block}{\cmd{np.loadtxt()}}
Ideal for clean, pure numerical matrices with no missing values.
\begin{lstlisting}[style=py]
import numpy as np
# Ingest homogeneous numeric matrix
arr = np.loadtxt("matrix.csv", 
                 delimiter=",", 
                 skiprows=1)
print(arr.shape)
\end{lstlisting}
\end{block}
\column{0.48\linewidth}
\begin{block}{\cmd{np.genfromtxt()}}
Robust parsing with missing value imputation.
\begin{lstlisting}[style=py]
# Replace missing cells with 0.0
arr_gen = np.genfromtxt(
    "numeric_sample.csv",
    delimiter=",",
    skip_header=1,
    filling_values=0.0
)
\end{lstlisting}
\end{block}
\end{columns}
\vspace{0.5em}
Export array to CSV: \texttt{np.savetxt("out.csv", arr, delimiter=",", fmt="\%0.2f")}.
\end{frame}

\begin{frame}[fragile]{Essential Parameters of Pandas \cmd{read_csv()}}
\small
\begin{lstlisting}[style=py]
import pandas as pd

df = pd.read_csv(
    "data/customers.csv",
    sep=",",                         # Column delimiter character (default ',')
    index_col="customer_id",         # Designate primary index column
    usecols=["customer_id", "full_name", "signup_date", "total_spent"], # Save RAM
    dtype={"customer_id": str},      # Explicit type enforcement
    parse_dates=["signup_date"],     # Parse ISO date strings to datetime64
    na_values=["NA", "null", "-"],   # Sentinel missing values
    encoding="utf-8"                 # Universal text encoding
)
\end{lstlisting}
\vspace{-0.3em}
\begin{alertblock}{Production Export Tip}
Use \cmd{df.to_csv("clean.csv", index=False, encoding="utf-8-sig")}. The \cmd{utf-8-sig} parameter includes a Byte Order Mark (BOM) enabling Microsoft Excel to automatically render UTF-8 characters properly.
\end{alertblock}
\end{frame}

\begin{frame}[fragile]{Out-of-Core Processing with \cmd{chunksize}}
\small
When datasets exceed available physical RAM, loading the entire file raises a fatal \key{MemoryError}. Chunk streaming processes records iteratively:

\begin{lstlisting}[style=py]
import pandas as pd

# Streaming iterator loading 100,000 rows per batch
chunk_iter = pd.read_csv("large_transactions.csv", chunksize=100000,
                         usecols=["store_id", "revenue"])

store_totals = {}
for i, chunk in enumerate(chunk_iter):
    # Process each batch independently in RAM
    grouped = chunk.groupby("store_id")["revenue"].sum()
    for sid, rev in grouped.items():
        store_totals[sid] = store_totals.get(sid, 0) + rev
    print(f"-> Finished chunk #{i+1}")

print("Total aggregated revenue:", store_totals)
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Flattening Nested JSON Hierarchies (\cmd{json_normalize})}
\small
RESTful API payloads commonly arrive with complex multi-level nesting (\key{Sub-dictionaries \& Arrays}).

\begin{lstlisting}[style=py]
import json, pandas as pd

with open("ecommerce_transactions.json", "r", encoding="utf-8") as f:
    orders = json.load(f)

# Unpack item arrays while propagating top-level order metadata
df_flat = pd.json_normalize(
    orders,
    record_path=["items"],                  # Explode rows per item element
    meta=["order_id", "order_date",         # Order-level attributes
          ["customer", "customer_id"],      # Nested sub-dictionary keys
          ["customer", "name"], 
          ["customer", "city"]]
)
print(df_flat.head(3))
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Relational Databases: SQLite \& Pandas}
\small
\begin{lstlisting}[style=py]
import sqlite3, pandas as pd

conn = sqlite3.connect("data/company.db")

# 1. Direct multi-table JOIN query into Pandas DataFrame
query = (
    "SELECT e.emp_id, e.full_name, d.dept_name, SUM(s.revenue) AS total_revenue "
    "FROM employees e "
    "JOIN departments d ON e.dept_id = d.dept_id "
    "JOIN monthly_sales s ON e.emp_id = s.emp_id "
    "GROUP BY e.emp_id, e.full_name, d.dept_name "
    "ORDER BY total_revenue DESC;"
)
df_sales = pd.read_sql_query(query, conn)

# 2. Persist transformed results back to SQLite
df_sales.to_sql("top_sales_summary", conn, if_exists="replace", index=False)
conn.close()
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Document NoSQL Database: MongoDB (\cmd{pymongo})}
\small
\begin{lstlisting}[style=py]
from pymongo import MongoClient
import pandas as pd

# Connect to database instance
client = MongoClient("mongodb://localhost:27017/")
db = client["retail_store_db"]
orders_col = db["customer_orders"]

# 1. Flexible document insertion
orders_col.insert_one({"order_id": "ORD-101", "items": ["P01", "P02"], "amount": 250.00})

# 2. Filter collection stream and normalize to DataFrame
cursor = orders_col.find({"amount": {"$gte": 100.00}})
df_orders = pd.json_normalize(list(cursor))

# Cleanse special BSON ObjectId identifier
if "_id" in df_orders.columns:
    df_orders["_id"] = df_orders["_id"].astype(str)
\end{lstlisting}

\begin{alertblock}{Production Security \& Performance Principles}
1. Always use parameterized queries (\cmd{?}) to protect against SQL Injection.  
2. Manage file and connection handles with \cmd{with} statements.  
3. Restrict columns with \cmd{usecols} and stream large tables via \cmd{chunksize}.
