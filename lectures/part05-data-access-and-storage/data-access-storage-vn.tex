\section{Tổng quan về truy xuất và lưu trữ dữ liệu}

\begin{frame}[fragile]{Truy xuất dữ liệu là gì?}
\textbf{Truy xuất dữ liệu} là quá trình lấy dữ liệu từ một nguồn lưu trữ và đưa vào môi trường phân tích.

\smallgap
Ví dụ: đọc tệp CSV vào Pandas.
\begin{lstlisting}[style=textstyle]
customers.csv
      |
      v
pd.read_csv()
      |
      v
DataFrame
\end{lstlisting}

Ví dụ: truy vấn database trước khi phân tích.
\begin{lstlisting}[style=textstyle]
SQL Database -> SQL Query -> Pandas DataFrame
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Lưu trữ dữ liệu là gì?}
\textbf{Lưu trữ dữ liệu} là quá trình ghi dữ liệu hoặc kết quả phân tích vào một định dạng hay hệ thống có thể tái sử dụng.

\begin{lstlisting}[style=textstyle]
DataFrame -> to_csv() -> report.csv
DataFrame -> to_sql() -> Database Table
\end{lstlisting}

\smallgap
Lưu trữ không chỉ là bước xuất kết quả. Nó còn hỗ trợ tái lập phân tích, báo cáo, chia sẻ dữ liệu và xây dựng mô hình sau này.
\end{frame}

\begin{frame}{Các nguồn dữ liệu phổ biến}
\begin{tabularx}{\textwidth}{lX}
\toprule
\textbf{Nhóm nguồn dữ liệu} & \textbf{Ví dụ} \\
\midrule
Tệp phẳng & CSV, TSV \\
Bảng tính & Excel \\
Dữ liệu bán cấu trúc & JSON \\
Dữ liệu Web & bảng HTML \\
Tài liệu & báo cáo PDF \\
Cơ sở dữ liệu quan hệ & SQLite, PostgreSQL, MySQL \\
Cơ sở dữ liệu NoSQL & MongoDB \\
\bottomrule
\end{tabularx}
\end{frame}

\begin{frame}[fragile]{Dữ liệu có cấu trúc và bán cấu trúc}

\begin{lstlisting}[style=textstyle]
CustomerID | Name | City | Revenue
\end{lstlisting}

Dữ liệu bán cấu trúc có thể chứa đối tượng và danh sách lồng nhau.
\begin{lstlisting}[style=pythonstyle]
{
    "customer": {"name": "An", "city": "Hanoi"},
    "orders": [
        {"product": "A"},
        {"product": "B"}
    ]
}
\end{lstlisting}
\end{frame}

\section{Làm việc với CSV bằng NumPy}

\begin{frame}[fragile]{Đọc dữ liệu số với \code{np.loadtxt()}}
\code{np.loadtxt()} phù hợp khi dữ liệu chủ yếu là số, có cấu trúc đơn giản và không có giá trị thiếu.

\begin{lstlisting}[style=pythonstyle]
import numpy as np

data = np.loadtxt(
    "matrix_data.csv",
    delimiter=",",
    skiprows=1
)

print(data)
print(data.shape)
\end{lstlisting}

Có thể tính trung bình từng cột bằng:
\begin{lstlisting}[style=pythonstyle]
print(np.mean(data, axis=0))
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Bài tập: \code{np.loadtxt()}}
Hoàn thiện chương trình để đọc \code{sales.csv}, trong đó các giá trị được phân cách bằng dấu phẩy.

\begin{lstlisting}[style=pythonstyle]
# data = np.loadtxt(
#     "sales.csv",
#     delimiter=...
# )

# print(data.shape)
\end{lstlisting}

Sau đó xác định số hàng và số cột của dữ liệu.
\end{frame}

\begin{frame}[fragile]{Xử lý giá trị thiếu với \code{np.genfromtxt()}}
\code{np.genfromtxt()} linh hoạt hơn \code{np.loadtxt()} khi tệp số chứa giá trị thiếu.

\begin{lstlisting}[style=pythonstyle]
data = np.genfromtxt(
    "numeric_data.csv",
    delimiter=",",
    skip_header=1,
    filling_values=0.0,
    dtype=float
)

print(data)
\end{lstlisting}

Trong ví dụ này, các giá trị thiếu được thay bằng \code{0.0}.
\end{frame}

\begin{frame}[fragile]{Ghi mảng với \code{np.savetxt()}}
\code{np.savetxt()} dùng để lưu mảng NumPy ra tệp văn bản.

\begin{lstlisting}[style=pythonstyle]
output = np.random.randn(5, 3)

np.savetxt(
    "output.csv",
    output,
    delimiter=",",
    fmt="%.2f",
    header="A,B,C",
    comments=""
)
\end{lstlisting}

\code{fmt="\%.2f"} ghi mỗi số với hai chữ số sau dấu thập phân.
\end{frame}

\begin{frame}{Các hàm CSV chính trong NumPy}
\begin{tabularx}{\textwidth}{lX}
\toprule
\textbf{Hàm} & \textbf{Công dụng} \\
\midrule
\code{np.loadtxt()} & Đọc dữ liệu số từ tệp văn bản có cấu trúc đơn giản \\
\code{np.genfromtxt()} & Đọc dữ liệu số và hỗ trợ giá trị thiếu \\
\code{np.savetxt()} & Ghi mảng NumPy ra tệp văn bản \\
\bottomrule
\end{tabularx}
\end{frame}

\begin{frame}[fragile]{Bài thực hành: Monthly Sales}
Tạo \code{monthly\_sales.csv} với nội dung:
\begin{lstlisting}[style=textstyle]
Month,Revenue,Cost
1,120,80
2,150,
3,180,110
4,,130
\end{lstlisting}

Thực hiện:
\begin{enumerate}
  \item Đọc dữ liệu bằng \code{np.genfromtxt()}.
  \item Thay giá trị thiếu bằng \code{0}.
  \item Tính trung bình Revenue và Cost.
  \item Lưu dữ liệu sau xử lý bằng \code{np.savetxt()}.
\end{enumerate}
\end{frame}

\section{Làm việc với CSV bằng Pandas}

\begin{frame}[fragile]{Đọc CSV với \code{pd.read\_csv()}}
Pandas cung cấp \code{pd.read\_csv()} để đọc dữ liệu dạng bảng từ tệp CSV.

\begin{lstlisting}[style=pythonstyle]
import pandas as pd

df = pd.read_csv("customers.csv")

print(df.head())
\end{lstlisting}

Pandas phù hợp khi dữ liệu chứa văn bản, ngày tháng, nhiều kiểu dữ liệu, giá trị thiếu và tên cột.
\end{frame}

\begin{frame}[fragile]{Xác định dấu phân cách với \code{sep}}
Không phải mọi tệp đều dùng dấu phẩy làm ký tự phân cách.

Tệp phân cách bằng dấu chấm phẩy:
\begin{lstlisting}[style=pythonstyle]
df = pd.read_csv(
    "data.csv",
    sep=";"
)
\end{lstlisting}

Tệp TSV:
\begin{lstlisting}[style=pythonstyle]
df = pd.read_csv(
    "data.tsv",
    sep="\t"
)
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Chỉ đọc cột cần thiết với \code{usecols}}
\code{usecols} cho phép chọn các cột cần đọc ngay khi nhập dữ liệu.

\begin{lstlisting}[style=pythonstyle]
df = pd.read_csv(
    "customers.csv",
    usecols=[
        "customer_id",
        "city",
        "total_spent"
    ]
)
\end{lstlisting}

Cách này giúp giảm RAM, rút ngắn thời gian đọc và loại bỏ sớm các biến không cần thiết.
\end{frame}

\begin{frame}[fragile]{Quy định kiểu dữ liệu với \code{dtype}}
Pandas có thể tự suy luận kiểu dữ liệu, nhưng nhiều trường hợp cần quy định tường minh.

\begin{lstlisting}[style=pythonstyle]
df = pd.read_csv(
    "customers.csv",
    dtype={
        "customer_id": str,
        "age": "Int64"
    }
)
\end{lstlisting}

Các trường ID thường là mã định danh, không phải đại lượng số học.
\end{frame}

\begin{frame}[fragile]{Chuyển ngày tháng với \code{parse\_dates}}
Có thể yêu cầu Pandas chuyển cột ngày tháng ngay khi đọc dữ liệu.

\begin{lstlisting}[style=pythonstyle]
df = pd.read_csv(
    "orders.csv",
    parse_dates=["OrderDate"]
)

print(df.dtypes)
\end{lstlisting}

Cách này hữu ích khi cần lọc, sắp xếp hoặc tính toán theo thời gian.
\end{frame}

\begin{frame}[fragile]{Xác định giá trị thiếu với \code{na\_values}}
Trong dữ liệu thực tế, giá trị thiếu có thể được biểu diễn bằng nhiều ký hiệu khác nhau.

\begin{lstlisting}[style=pythonstyle]
df = pd.read_csv(
    "customers.csv",
    na_values=[
        "NA",
        "N/A",
        "-",
        "missing"
    ]
)
\end{lstlisting}

Pandas sẽ chuyển các ký hiệu này thành giá trị thiếu chuẩn.
\end{frame}

\begin{frame}[fragile]{Mã hóa ký tự}
Khi làm việc với dữ liệu tiếng Việt, UTF-8 là lựa chọn phổ biến.

\begin{lstlisting}[style=pythonstyle]
df = pd.read_csv(
    "customers.csv",
    encoding="utf-8"
)
\end{lstlisting}

Khi xuất CSV để mở bằng Excel trên Windows:
\begin{lstlisting}[style=pythonstyle]
df.to_csv(
    "customers_clean.csv",
    index=False,
    encoding="utf-8-sig"
)
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Kiểm tra dữ liệu sau khi nhập}
Sau khi đọc dữ liệu, cần kiểm tra ngay cấu trúc và chất lượng của DataFrame.

\begin{lstlisting}[style=pythonstyle]
df.head()
df.shape
df.info()
df.dtypes
df.isna().sum()
\end{lstlisting}

Các lệnh này giúp kiểm tra số hàng, số cột, tên cột, kiểu dữ liệu và số giá trị thiếu.
\end{frame}

\begin{frame}[fragile]{Đọc dữ liệu lớn với \code{chunksize}}
Khi tệp CSV lớn hơn bộ nhớ khả dụng, không nên đọc toàn bộ dữ liệu cùng lúc.

\begin{lstlisting}[style=pythonstyle]
total = 0

for chunk in pd.read_csv(
    "transactions.csv",
    chunksize=100000
):
    total += chunk["Amount"].sum()

print(total)
\end{lstlisting}

\code{chunksize} cho phép xử lý dữ liệu tuần tự theo từng khối.
\end{frame}

\begin{frame}[fragile]{Ghi DataFrame ra CSV}
Sử dụng phương thức \code{to\_csv()}.

\begin{lstlisting}[style=pythonstyle]
df.to_csv(
    "clean_data.csv",
    index=False,
    encoding="utf-8-sig"
)
\end{lstlisting}

\code{index=False} giúp không ghi chỉ mục của DataFrame thành một cột bổ sung.
\end{frame}

\begin{frame}{Các hàm và tham số CSV chính của Pandas}
\begin{tabularx}{\textwidth}{lX}
\toprule
\textbf{Hàm hoặc tham số} & \textbf{Công dụng} \\
\midrule
\code{pd.read\_csv()} & Đọc CSV hoặc tệp văn bản có dấu phân cách \\
\code{usecols=} & Chỉ đọc các cột cần thiết \\
\code{dtype=} & Quy định kiểu dữ liệu \\
\code{parse\_dates=} & Chuyển cột sang kiểu ngày tháng \\
\code{na\_values=} & Xác định ký hiệu giá trị thiếu \\
\code{chunksize=} & Đọc dữ liệu theo từng khối \\
\code{df.to\_csv()} & Ghi DataFrame ra CSV \\
\bottomrule
\end{tabularx}
\end{frame}

\begin{frame}{Kiểm tra kiến thức: Pandas CSV}

  \item \code{split=True}

\begin{frame}{Bài thực hành: Customer CSV}
Tạo \code{customers.csv} gồm các cột: CustomerID, Name, City, Age, SignupDate và TotalSpent.

  \item Kiểm tra bằng \code{head()}, \code{shape}, \code{info()} và \code{isna().sum()}.

\end{frame}

\section{Làm việc với Microsoft Excel}

\begin{frame}[fragile]{Đọc một trang tính}
Pandas sử dụng \code{pd.read\_excel()} để đọc dữ liệu Excel.

\begin{lstlisting}[style=pythonstyle]
df = pd.read_excel(
    "products.xlsx",
    sheet_name="Products",
    engine="openpyxl"
)
\end{lstlisting}

\code{sheet\_name} xác định trang tính cần đọc.
\end{frame}

\begin{frame}[fragile]{Đọc toàn bộ trang tính}
Có thể đọc toàn bộ workbook bằng \code{sheet\_name=None}.

\begin{lstlisting}[style=pythonstyle]
workbook = pd.read_excel(
    "products.xlsx",
    sheet_name=None
)

print(workbook.keys())
\end{lstlisting}

Kết quả là dictionary: key là tên trang tính, value là DataFrame tương ứng.
\end{frame}

\begin{frame}[fragile]{Ghi nhiều DataFrame vào một workbook}

\section{Làm việc với JSON}

\begin{frame}[fragile]{JSON là gì?}
JSON là định dạng dữ liệu phổ biến trong Web API, thương mại điện tử, microservices và cơ sở dữ liệu NoSQL.

\begin{lstlisting}[style=pythonstyle]
[
    {"customer_id": "C001", "city": "Hanoi"},
    {"customer_id": "C002", "city": "Danang"}
]
\end{lstlisting}

Với JSON tương đối phẳng, có thể dùng:
\begin{lstlisting}[style=pythonstyle]
df = pd.read_json("customers.json")
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{JSON lồng nhau}
Dữ liệu JSON thực tế thường có nhiều cấp.

\begin{lstlisting}[style=pythonstyle]
{
    "order_id": "O001",
    "customer": {
        "customer_id": "C001",
        "city": "Hanoi"
    },
    "items": [
        {"sku": "P001", "qty": 2},
        {"sku": "P002", "qty": 1}
    ]
}
\end{lstlisting}

Để phân tích bằng Pandas, thường cần chuyển cấu trúc này thành bảng.
\end{frame}

\begin{frame}[fragile]{Làm phẳng với \code{pd.json\_normalize()}}
\begin{lstlisting}[style=pythonstyle]
df_items = pd.json_normalize(
    orders,
    record_path=["items"],
    meta=[
        "order_id",
        ["customer", "customer_id"],
        ["customer", "city"]
    ]
)
\end{lstlisting}

\begin{itemize}
  \item \code{record\_path}: danh sách lồng nhau cần mở rộng thành các dòng.
  \item \code{meta}: các thuộc tính ở cấp cha cần giữ lại.
\end{itemize}
\end{frame}

\begin{frame}{Bài tập và kiểm tra kiến thức về JSON}
\textbf{Bài tập.} Tạo dữ liệu gồm ba đơn hàng. Mỗi đơn hàng có Customer và Items. Dùng \code{pd.json\_normalize()} để mỗi item trở thành một dòng.

\smallgap
\textbf{Câu hỏi.} Vai trò của \code{record\_path} là gì?
\begin{enumerate}[A.]
  \item Xác định danh sách lồng nhau cần mở rộng thành các dòng
  \item Đổi tên cột
  \item Ghi dữ liệu ra JSON
  \item Xóa giá trị thiếu
\end{enumerate}
\end{frame}

\section{Làm việc với SQLite}

\begin{frame}[fragile]{SQLite là gì?}
SQLite là hệ quản trị cơ sở dữ liệu quan hệ gọn nhẹ. Toàn bộ cơ sở dữ liệu có thể lưu trong một tệp.

\begin{lstlisting}[style=textstyle]
company.db
\end{lstlisting}

Python hỗ trợ SQLite thông qua module chuẩn \code{sqlite3}.
\begin{lstlisting}[style=pythonstyle]
import sqlite3

conn = sqlite3.connect("company.db")
cursor = conn.cursor()
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Thực hiện truy vấn SQL}
\begin{lstlisting}[style=pythonstyle]
cursor.execute(
    """
    SELECT emp_id, full_name, salary
    FROM employees
    WHERE salary >= ?
    """,
    (30000000,)
)

rows = cursor.fetchall()
\end{lstlisting}

\code{fetchall()} trả về toàn bộ các dòng thỏa mãn truy vấn.
\end{frame}

\begin{frame}[fragile]{Truy vấn có tham số}
Khi giá trị truy vấn đến từ biến Python hoặc dữ liệu người dùng, nên sử dụng truy vấn có tham số.

\begin{lstlisting}[style=pythonstyle]
cursor.execute(
    "SELECT * FROM users WHERE id = ?",
    (user_id,)
)
\end{lstlisting}

Cách này tách cấu trúc SQL khỏi giá trị được truyền vào và giúp giảm rủi ro SQL Injection.
\end{frame}

\begin{frame}[fragile]{Đọc kết quả SQL vào Pandas}
Pandas có thể đọc trực tiếp kết quả của một truy vấn SQL.

\begin{lstlisting}[style=pythonstyle]
query = """
SELECT
    emp_id,
    full_name,
    salary
FROM employees
"""

df = pd.read_sql_query(
    query,
    conn
)
\end{lstlisting}

Kết quả được trả về dưới dạng DataFrame.
\end{frame}

\begin{frame}[fragile]{Kết hợp nhiều bảng bằng JOIN}
\begin{lstlisting}[style=pythonstyle]
query = """
SELECT
    e.emp_id,
    e.full_name,
    d.dept_name
FROM employees e
LEFT JOIN departments d
ON e.dept_id = d.dept_id;
"""

df = pd.read_sql_query(query, conn)
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Ghi DataFrame vào SQLite}
Pandas cung cấp phương thức \code{to\_sql()}.

\begin{lstlisting}[style=pythonstyle]
df.to_sql(
    name="sales_summary",
    con=conn,
    if_exists="replace",
    index=False
)
\end{lstlisting}

Các lựa chọn thường dùng của \code{if\_exists}: \code{fail}, \code{replace}, \code{append}.
\end{frame}

\begin{frame}{Các hàm SQLite và bài tập}
\begin{tabularx}{\textwidth}{lX}
\toprule
\textbf{Hàm} & \textbf{Công dụng} \\
\midrule
\code{sqlite3.connect()} & Mở kết nối đến SQLite \\
\code{cursor.execute()} & Thực thi câu lệnh SQL \\
\code{fetchall()} & Lấy kết quả truy vấn \\
\code{pd.read\_sql\_query()} & Chuyển kết quả SQL thành DataFrame \\
\code{df.to\_sql()} & Ghi DataFrame vào bảng SQL \\
\bottomrule
\end{tabularx}

\smallgap
\textbf{Bài tập.} Tạo \code{sales.db}, JOIN \code{orders} với \code{customers}, tính tổng Amount theo City và lưu kết quả vào \code{city\_sales\_summary}.
\end{frame}

\section{Làm việc với MongoDB và PyMongo}

\begin{frame}[fragile]{MongoDB là gì?}
MongoDB là hệ quản trị cơ sở dữ liệu NoSQL hướng document.

\begin{columns}[T]
\begin{column}{0.45\textwidth}
Cơ sở dữ liệu quan hệ:
\begin{lstlisting}[style=textstyle]
Table
  -> Row
\end{lstlisting}
\end{column}
\begin{column}{0.45\textwidth}
MongoDB:
\begin{lstlisting}[style=textstyle]
Collection
  -> Document
\end{lstlisting}
\end{column}
\end{columns}

\smallgap
Document có thể chứa giá trị đơn, đối tượng lồng nhau, danh sách và nhiều cấp cấu trúc.
\end{frame}

\begin{frame}[fragile]{Kết nối MongoDB}
PyMongo cung cấp \code{MongoClient} để kết nối MongoDB.

\begin{lstlisting}[style=pythonstyle]
from pymongo import MongoClient

client = MongoClient(
    "mongodb://localhost:27017/"
)

db = client["retail_db"]
orders = db["orders"]
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Các thao tác CRUD}
CRUD gồm Create, Read, Update và Delete.

\begin{lstlisting}[style=pythonstyle]
# Create
orders.insert_one(sample_order)

# Read one document
orders.find_one({"order_id": "ORD001"})

# Filter multiple documents
orders.find({"total_amount": {"$gte": 1000000}})

# Update
orders.update_one(
    {"order_id": "ORD001"},
    {"$set": {"status": "REFUNDED"}}
)
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Chuyển dữ liệu MongoDB sang Pandas}
Kết quả truy vấn có thể được chuyển thành danh sách các document.

\begin{lstlisting}[style=pythonstyle]
documents = list(
    orders.find()
)

df = pd.json_normalize(
    documents
)
\end{lstlisting}

Trường \code{\_id} có thể được chuyển sang chuỗi trước khi xuất dữ liệu.
\begin{lstlisting}[style=pythonstyle]
df["_id"] = df["_id"].astype(str)
\end{lstlisting}
\end{frame}

\begin{frame}{Bài tập và kiểm tra kiến thức về MongoDB}
\textbf{Bài tập.} Tạo collection \code{customer\_orders} có ít nhất năm document. Thêm đơn hàng, truy vấn các đơn hàng đã thanh toán, cập nhật trạng thái một đơn hàng, chuyển kết quả sang Pandas và tính doanh thu đã thanh toán.

\smallgap
\textbf{Câu hỏi.} Hàm Pandas nào hữu ích với document MongoDB lồng nhau?
\begin{enumerate}[A.]
  \item \code{pd.json\_normalize()}
  \item \code{pd.read\_csv()}
  \item \code{pd.crosstab()}
  \item \code{pd.cut()}
\end{enumerate}
\end{frame}
