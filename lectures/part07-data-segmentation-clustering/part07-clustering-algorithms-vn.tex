% Slide 4: Hard vs Soft Clustering
\begin{frame}{Phân loại Hình thức: Hard Clustering vs Soft Clustering}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{1. Phân cụm cứng (Hard Clustering):}
      \begin{itemize}
        \item Mỗi điểm $x_i$ thuộc duy nhất về một cụm $C_k$ với xác suất tuyệt đối $100\%$.
        \item Nhãn phân cụm nhị phân: $z_{ik} \in \{0, 1\}$.
        \item \textit{Đại diện tiêu biểu:} K-Means, Hierarchical, DBSCAN.
      \end{itemize}
      \vspace{0.5em}
      \textbf{2. Phân cụm mềm (Soft / Fuzzy Clustering):}
      \begin{itemize}
        \item Mỗi điểm thuộc về tất cả các cụm với một \textbf{trọng số xác suất} $\gamma_{ik} \in [0, 1]$ thỏa mãn $\sum_{k=1}^K \gamma_{ik} = 1$.
        \item \textit{Đại diện tiêu biểu:} GMM (EM), Fuzzy C-Means.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{center}
        \includegraphics[width=0.98\linewidth]{images/hard-vs-soft-clustering.png}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

% Slide 6: Centroid-based - K-Means
\begin{frame}{2. Centroid-based: Thuật toán K-Means Cổ điển}
  \small
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Quy trình 4 bước của Lloyd (1957):}
      \begin{enumerate}\setlength{\itemsep}{1pt}
        \item \textbf{Khởi tạo:} Chọn ngẫu nhiên $K$ tâm cụm $\mu_1, \dots, \mu_K$.
        \item \textbf{Gán cụm (Assignment):} Gán mỗi điểm $x_i$ vào cụm có tâm gần nhất:
          $$c^{(i)} = \arg\min_k \|x_i - \mu_k\|_2^2$$
        \item \textbf{Cập nhật tâm (Update):} Tính lại toạ độ trọng tâm:
          $$\mu_k = \frac{1}{|C_k|} \sum_{x_i \in C_k} x_i$$
        \item \textbf{Lặp lại} bước 2 và 3 cho đến khi tâm không đổi.
      \end{enumerate}
      \textbf{Hàm mục tiêu (WCSS / Inertia):}
      $$J = \sum_{k=1}^K \sum_{x_i \in C_k} \|x_i - \mu_k\|_2^2 \to \min$$
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{center}
        \includegraphics[width=0.95\linewidth,height=0.62\textheight,keepaspectratio]{images/centroid-based-kmeans.png}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

% Slide 7: K-Means++ và Elbow Method
\begin{frame}{Centroid-based: K-Means++ và Phương pháp Khuỷu tay}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{K-Means++ (Arthur \& Vassilvitskii, 2007):}
      \begin{itemize}
        \item Khởi tạo tâm ngẫu nhiên dễ rơi vào \textbf{cực tiểu cục bộ kém}.
        \item K-Means++ chọn tâm tiếp theo với xác suất tỷ lệ thuận với bình phương khoảng cách tới tâm gần nhất:
          $$P(x) = \frac{D(x)^2}{\sum_{x' \in \mathcal{X}} D(x')^2}$$
        \item \textbf{Đảm bảo lý thuyết:} Kỳ vọng nghiệm đạt xấp xỉ $O(\log K)$ so với tối ưu toàn cục.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Phương pháp Khuỷu tay (Elbow Method):}
      \begin{center}
        \includegraphics[width=0.92\linewidth]{images/elbow-method-curve.png}
      \end{center}
      \begin{itemize}
        \item Vẽ đồ thị Inertia (WCSS) theo số cụm $K$.
        \item Chọn $K$ tại \textbf{điểm gãy (Elbow)} nơi tốc độ giảm WCSS bắt đầu chững lại rõ rệt.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% Slide 8: Connectivity-based - Hierarchical
\begin{frame}{3. Connectivity-based: Phân cụm Phân cấp (Hierarchical)}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Hai hướng tiếp cận đối xứng:}
      \begin{itemize}
        \item \textbf{Agglomerative (Bottom-Up):} Đi từ $N$ cụm đơn lẻ $\to$ liên tục sáp nhập 2 cụm gần nhất $\to$ 1 cụm tổng thể ($N-1$ bước).
        \item \textbf{Divisive (Top-Down):} Đi từ 1 cụm lớn $\to$ liên tục phân tách đệ quy $\to$ $N$ cụm đơn lẻ.
      \end{itemize}
      \vspace{0.3em}
      \textbf{Ưu điểm đột phá:}
      \begin{itemize}
        \item \textbf{Không cần chọn trước số cụm $K$}.
        \item Cho phép phân tích cấu trúc đa tầng lồng nhau (Nested Hierarchy).
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}

  \end{columns}
\end{frame}

% Slide 9: Biểu đồ cây Dendrogram
\begin{frame}{Biểu đồ Cây Phân cấp (Dendrogram) \& Cắt Cây Tìm $K$}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{center}
        \includegraphics[width=0.98\linewidth]{images/longest-path-dendrogram.png}
      \end{center}
    \end{column}
    \begin{column}{0.45\textwidth}
      \textbf{Quy tắc Cắt cây tìm $K$ tối ưu:}
      \begin{enumerate}
        \item Tìm \textbf{nhánh thẳng đứng dài nhất} mà không bị bất kỳ đường ngang nào cắt qua.
        \item Vẽ một đường cắt nằm ngang xuyên qua khoảng trống này.
        \item \textbf{Số lượng cụm $K$ tối ưu} chính là số lượng nhánh dọc bị đường thẳng này cắt qua!
      \end{enumerate}
      \vspace{0.3em}
      \textit{Ý nghĩa:} Khoảng cách dọc lớn chứng tỏ cấu trúc cụm giữ nguyên trạng thái phân tách ổn định trên một khoảng khoảng cách rộng.
    \end{column}
  \end{columns}
\end{frame}

% Slide 10: 5 Tiêu chuẩn Liên kết (Linkage Criteria)
\begin{frame}{Các Tiêu chuẩn Liên kết Cụm (Linkage Criteria)}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{center}
        \includegraphics[width=0.95\linewidth]{images/distance-metrics.png}
      \end{center}
    \end{column}
    \begin{column}{0.52\textwidth}
      \scriptsize
      \begin{itemize}
        \item \textbf{Single Linkage (Min):} $\min d(x, z)$. Nhận diện cụm phi cầu tốt, nhưng dễ mắc \textbf{Hiện tượng Nối chuỗi (Chaining Effect)}.
        \item \textbf{Complete Linkage (Max):} $\max d(x, z)$. Tạo cụm tròn đặc, ngăn nối chuỗi nhưng nhạy cảm với ngoại lai.
        \item \textbf{Average Linkage (UPGMA):} Trung bình cộng khoảng cách cặp đôi. Cân bằng, đạt hệ số Cophenetic (CPCC) cao.
        \item \textbf{Centroid Linkage:} $\|\mu_A - \mu_B\|$. Dễ bị \textbf{Hiện tượng Đảo ngược (Inversion)} phá vỡ tính đơn điệu.
        \item \textbf{Ward's Method:} Tối thiểu hóa mức tăng tổng phương sai nội cụm $\Delta\text{WCSS}$. Tạo cụm đều, đẹp nhất cho dữ liệu chuẩn.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% Slide 11: Density-based - DBSCAN
\begin{frame}{4. Density-based: Thuật toán DBSCAN}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Hai siêu tham số:} Bán kính $\epsilon$ và $\text{MinPts}$.
      \vspace{0.3em}
      
      \textbf{Phân loại 3 nhóm điểm:}
      \begin{itemize}
        \item \textbf{Core Point:} $|N_\epsilon(p)| \ge \text{MinPts}$.
        \item \textbf{Border Point:} $|N_\epsilon(p)| < \text{MinPts}$ nhưng nằm trong bán kính $\epsilon$ của một Core Point.
        \item \textbf{Noise Point:} Không thuộc 2 nhóm trên (gán nhãn \textbf{-1}).
      \end{itemize}
      \vspace{0.3em}
      \textbf{Cơ chế lan truyền cụm:}
      Liên thông mật độ thông qua các chuỗi điểm tiếp cận trực tiếp.

% Slide 13: Distribution-based - GMM & EM
\begin{frame}{5. Distribution-based: Gaussian Mixture Models (GMM)}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Mô hình xác suất tạo sinh (Generative Model):}
      $$p(x) = \sum_{k=1}^K \pi_k \mathcal{N}(x \mid \mu_k, \Sigma_k)$$
      \begin{itemize}\setlength{\itemsep}{1pt}
        \item $\pi_k$: Trọng số thành phần ($\sum \pi_k = 1$).
        \item $\mu_k$: Vector kỳ vọng (tâm cụm).
        \item $\Sigma_k$: Ma trận hiệp phương sai (hình dạng elip).
      \end{itemize}
      \textbf{Phân cụm mềm (Soft Assignment):}
      $$\gamma_{nk} = P(z_{nk}=1 \mid x_n) = \frac{\pi_k \mathcal{N}(x_n \mid \mu_k, \Sigma_k)}{\sum_{j=1}^K \pi_j \mathcal{N}(x_n \mid \mu_j, \Sigma_j)}$$
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{center}
        \includegraphics[width=0.92\linewidth,height=0.62\textheight,keepaspectratio]{images/gmm-distribution-concept.png}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

% Slide 14: Thuật toán EM và 4 Covariance Types
\begin{frame}{Thuật toán Expectation-Maximization (EM) \& Ma trận Hiệp phương sai}
  \small
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Hai bước luân phiên của EM:}
      \begin{itemize}\setlength{\itemsep}{1pt}
        \item \textbf{E-step:} Tính ma trận trách nhiệm $\gamma_{nk}$ theo tham số hiện tại.
        \item \textbf{M-step:} Cập nhật tham số theo trung bình có trọng số:
          $$\mu_k = \frac{1}{N_k}\sum \gamma_{nk}x_n, \quad \pi_k = \frac{N_k}{N}$$
          $$\Sigma_k = \frac{1}{N_k}\sum \gamma_{nk}(x_n - \mu_k)(x_n - \mu_k)^T$$
      \end{itemize}
      \textbf{Bảo đảm toán học:} Hàm Log-Likelihood luôn tăng đơn điệu sau mỗi epoch!
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{center}
        \includegraphics[width=0.95\linewidth,height=0.42\textheight,keepaspectratio]{images/gmm-covariance-types.png}
      \end{center}
      \scriptsize
      \textbf{4 Dạng ma trận hiệp phương sai:}
      \begin{itemize}\setlength{\itemsep}{0pt}
        \item \code{spherical}: Hình cầu, đẳng hướng.
        \item \code{diag}: Elip song song các trục tọa độ.
        \item \code{tied}: Các cụm chia sẻ chung 1 ma trận.
        \item \code{full}: Elip tự do mọi hướng (tổng quát).
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% Slide 15: Đánh giá mô hình phân cụm
\begin{frame}{6. Đánh giá Chất lượng Phân cụm (Clustering Metrics)}
  \small
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Khi KHÔNG có nhãn thực tế (Internal):}
      \begin{itemize}\setlength{\itemsep}{1pt}
        \item \textbf{Silhouette Coefficient ($s \in [-1, 1]$):}
          $$s(i) = \frac{b(i) - a(i)}{\max(a(i), b(i))}$$
          $a(i)$: khoảng cách nội cụm trung bình; $b(i)$: khoảng cách tới cụm gần nhất. $s > 0.5 \implies$ Tốt.
        \item \textbf{Davies-Bouldin Index (DBI):} Tỷ số giữa độ phân tán nội cụm và khoảng cách liên cụm. \textbf{Càng nhỏ càng tốt}.
        \item \textbf{Calinski-Harabasz Index:} Tỷ số phương sai liên cụm / nội cụm. \textbf{Càng lớn càng tốt}.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Khi CÓ nhãn đối chuẩn (External):}
      \begin{itemize}\setlength{\itemsep}{1pt}
        \item \textbf{Adjusted Rand Index (ARI):} Đo tỷ lệ cặp điểm được gán cùng nhóm, khử yếu tố ngẫu nhiên ($-1 \le \text{ARI} \le 1$).
        \item \textbf{Normalized Mutual Information (NMI):} Tương hỗ thông tin chuẩn hóa giữa nhãn thực và nhãn dự báo ($0 \le \text{NMI} \le 1$).
      \end{itemize}
      \vspace{0.2em}
      \textbf{Chọn mô hình xác suất (GMM):}
      $$\text{BIC} = -2\ln L + p\ln N \to \min$$
      $$\text{AIC} = -2\ln L + 2p \to \min$$
    \end{column}

            \item Kích thước dữ liệu $N$ (Scale).
            \item Hình thái hình học của cụm.
            \item Tỷ lệ nhiễu và ngoại lai.
            \item Yêu cầu bài toán kinh doanh (cứng hay mềm).
          \end{itemize}
      \end{itemize}
    \end{column}

            \item 1 chu kỳ E-step \& M-step trong GMM.
