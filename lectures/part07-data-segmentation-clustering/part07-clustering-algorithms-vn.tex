% Slide 4: Hard vs Soft Clustering
\begin{frame}{Phân loại Hình thức: Hard Clustering vs Soft Clustering}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{1. Phân cụm cứng (Hard Clustering):}
      \begin{itemize}\setlength{\itemsep}{2pt}
        \item Mỗi điểm $x_i$ thuộc duy nhất về một cụm $C_k$ với xác suất tuyệt đối $100\%$.
        \item Nhãn phân cụm nhị phân: $z_{ik} \in \{0, 1\}$ thỏa mãn $\sum_{k=1}^K z_{ik} = 1$.
        \item \textit{Đại diện tiêu biểu:} K-Means, Hierarchical, DBSCAN.
      \end{itemize}
      \vspace{0.4em}
      \textbf{2. Phân cụm mềm (Soft / Fuzzy Clustering):}
      \begin{itemize}\setlength{\itemsep}{2pt}
        \item Mỗi điểm thuộc về tất cả các cụm với một \textbf{trọng số xác suất} hoặc mức độ thuộc (membership) $\gamma_{ik} \in [0, 1]$ thỏa mãn $\sum_{k=1}^K \gamma_{ik} = 1$.
        \item \textit{Đại diện tiêu biểu:} GMM (EM), Fuzzy C-Means (FCM).
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{center}
        \includegraphics[width=0.98\linewidth,height=0.65\textheight,keepaspectratio]{images/hard-vs-soft-clustering.png}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

% Slide 5: Độ đo khoảng cách và độ tương đồng
\begin{frame}{Các Độ đo Khoảng cách \& Độ tương đồng (Distance Metrics)}
  \small
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Các hàm khoảng cách thông dụng:}
      \begin{itemize}\setlength{\itemsep}{2pt}
        \item \textbf{Euclidean Distance ($L_2$ norm):}
          $$d(x, z) = \sqrt{\sum_{j=1}^d (x_j - z_j)^2}$$
          Chuẩn mực cho biến định lượng liên tục cùng thang đo.
        \item \textbf{Manhattan Distance ($L_1$ norm / City Block):}
          $$d(x, z) = \sum_{j=1}^d |x_j - z_j|$$
          Bền vững hơn trước outliers, phù hợp không gian lưới.
        \item \textbf{Cosine Similarity (Góc vector):}
          $$\cos(\theta) = \frac{x \cdot z}{\|x\|_2 \|z\|_2}$$
          Chuẩn mực cho văn bản (TF-IDF), hệ gợi ý.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{center}
        \includegraphics[width=0.98\linewidth,height=0.42\textheight,keepaspectratio]{images/distance-metrics.png}
      \end{center}
      \footnotesize
      \textbf{Lưu ý sống còn:} Phải thực hiện \textbf{Chuẩn hóa dữ liệu (StandardScaler)} trước khi tính khoảng cách, nếu không biến có giá trị lớn sẽ áp đảo mô hình!
    \end{column}
  \end{columns}
\end{frame}

% Slide 7: Centroid-based - K-Means
\begin{frame}{2. Centroid-based: Thuật toán K-Means Cổ điển}
  \small
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Quy trình 4 bước của Lloyd (1957):}
      \begin{enumerate}\setlength{\itemsep}{1pt}
        \item \textbf{Khởi tạo:} Chọn ngẫu nhiên $K$ tâm cụm $\mu_1, \dots, \mu_K$.
        \item \textbf{Gán cụm (Assignment):} Gán mỗi điểm $x_i$ vào cụm có tâm gần nhất:
          $$c^{(i)} = \arg\min_k \|x_i - \mu_k\|_2^2$$
        \item \textbf{Cập nhật tâm (Update):} Tính lại toạ độ trọng tâm:
          $$\mu_k = \frac{1}{|C_k|} \sum_{x_i \in C_k} x_i$$
        \item \textbf{Lặp lại} bước 2 và 3 cho đến khi tâm không đổi.
      \end{enumerate}
      \textbf{Hàm mục tiêu (WCSS / Inertia):}
      $$J = \sum_{k=1}^K \sum_{x_i \in C_k} \|x_i - \mu_k\|_2^2 \to \min$$
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{center}
        \includegraphics[width=0.95\linewidth,height=0.62\textheight,keepaspectratio]{images/centroid-based-kmeans.png}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

% Slide 8: K-Means++ và Elbow Method
\begin{frame}{Centroid-based: K-Means++ và Phương pháp Khuỷu tay}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{K-Means++ (Arthur \& Vassilvitskii, 2007):}
      \begin{itemize}\setlength{\itemsep}{2pt}
        \item Khởi tạo tâm ngẫu nhiên dễ rơi vào \textbf{cực tiểu cục bộ kém}.
        \item K-Means++ chọn tâm tiếp theo với xác suất tỷ lệ thuận với bình phương khoảng cách tới tâm gần nhất:
          $$P(x) = \frac{D(x)^2}{\sum_{x' \in \mathcal{X}} D(x')^2}$$
        \item \textbf{Đảm bảo lý thuyết:} Kỳ vọng nghiệm đạt xấp xỉ $O(\log K)$ so với tối ưu toàn cục.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Phương pháp Khuỷu tay (Elbow Method):}
      \begin{center}
        \includegraphics[width=0.92\linewidth,height=0.45\textheight,keepaspectratio]{images/elbow-method-curve.png}
      \end{center}
      \begin{itemize}\setlength{\itemsep}{1pt}
        \item Vẽ đồ thị Inertia (WCSS) theo số cụm $K$.
        \item Chọn $K$ tại \textbf{điểm gãy (Elbow)} nơi tốc độ giảm WCSS bắt đầu chững lại rõ rệt.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% Slide 9: Centroid-based - K-Medoids (PAM)
\begin{frame}{Centroid-based: Biến thể Bền vững K-Medoids (PAM)}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Vấn đề của K-Means với Outliers:}
      \begin{itemize}\setlength{\itemsep}{2pt}
        \item Điểm trung bình cộng (Mean) cực kỳ nhạy cảm với ngoại lai, làm lệch tâm cụm nghiêm trọng.
      \end{itemize}
      \vspace{0.3em}
      \textbf{Giải pháp K-Medoids (Kaufman \& Rousseeuw, 1987):}
      \begin{itemize}\setlength{\itemsep}{2pt}
        \item Tâm cụm bắt buộc phải là \textbf{một điểm dữ liệu thực tế} trong tập dữ liệu (Medoid).
        \item Tối ưu hàm mất mát khoảng cách trị tuyệt đối ($L_1$ norm):
          $$J_{\text{medoid}} = \sum_{k=1}^K \sum_{x \in C_k} \|x - m_k\|_1 \to \min$$
        \item \textbf{Ưu điểm:} Cực kỳ kiên cường trước nhiễu và ngoại lai.
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{center}
        \includegraphics[width=0.92\linewidth,height=0.62\textheight,keepaspectratio]{images/kmeans-clusters-visualization.png}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

% Slide 10: Connectivity-based - Hierarchical
\begin{frame}{3. Connectivity-based: Phân cụm Phân cấp (Hierarchical)}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Hai hướng tiếp cận đối ngẫu:}
      \begin{enumerate}\setlength{\itemsep}{2pt}
        \item \textbf{Agglomerative (Tích tụ / Bottom-Up):}
          \begin{itemize}
            \item Bắt đầu: $N$ cụm đơn lẻ ($N$ lá).
            \item Mỗi bước: Gộp 2 cụm gần nhau nhất.
            \item Kết thúc: 1 siêu cụm duy nhất (Gốc).
          \end{itemize}
        \item \textbf{Divisive (Phân tách / Top-Down):}
          \begin{itemize}
            \item Bắt đầu: 1 cụm tổng thể chứa toàn bộ dữ liệu.
            \item Phân rã đệ quy thành các cụm nhỏ hơn.
          \end{itemize}
      \end{enumerate}
      \textbf{Ưu thế nổi bật:} Không cần chọn trước số lượng cụm $K$ khi dựng cây!
    \end{column}

\end{frame}

% Slide 11: Dendrogram và 5 Linkage Criteria
\begin{frame}{Biểu đồ Cây Dendrogram \& 5 Tiêu chuẩn Liên kết (Linkage)}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Giải phẫu Biểu đồ Dendrogram:}
      \begin{center}
        \includegraphics[width=0.95\linewidth,height=0.42\textheight,keepaspectratio]{images/longest-path-dendrogram.png}
      \end{center}
      \footnotesize
      Trục tung biểu thị khoảng cách sáp nhập. \textbf{Quy tắc cắt cụm:} Cắt ngang qua nhánh dọc dài nhất không bị ngắt quãng.
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{5 Tiêu chuẩn Liên kết (Linkage):}
      \begin{itemize}\setlength{\itemsep}{1pt}\footnotesize
        \item \textbf{Single Linkage:} $D(A, B) = \min d(x, y)$ \\
          (Dễ bị hiện tượng nối chuỗi - Chaining Effect).
        \item \textbf{Complete Linkage:} $D(A, B) = \max d(x, y)$ \\
          (Tạo cụm hình cầu đặc, tránh nối chuỗi).
        \item \textbf{Average Linkage:} $D(A, B) = \frac{1}{|A||B|}\sum d(x, y)$ \\
          (Cân bằng, ít nhạy cảm với ngoại lai).
        \item \textbf{Centroid Linkage:} $D(A, B) = \|\mu_A - \mu_B\|^2$ \\
          (Có rủi ro đảo ngược Inversion).
        \item \textbf{Ward's Method:} Tối thiểu hóa mức tăng tổng phương sai WCSS sau khi sáp nhập (\textbf{Tốt nhất thực tế}).
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% Slide 12: Density-based - DBSCAN
\begin{frame}{4. Density-based: Thuật toán DBSCAN}
  \small
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Hai siêu tham số cốt lõi:}
      \begin{itemize}
        \item $\mathbf{\epsilon}$ (Epsilon): Bán kính lân cận vùng khảo sát.
        \item $\mathbf{\text{MinPts}}$: Số lượng mẫu tối thiểu trong bán kính $\epsilon$.
      \end{itemize}
      \vspace{0.2em}
      \textbf{Phân loại 3 nhóm điểm:}
      \begin{itemize}
        \item \textbf{Core Point:} $|N_\epsilon(p)| \ge \text{MinPts}$.
        \item \textbf{Border Point:} $|N_\epsilon(p)| < \text{MinPts}$ nhưng nằm trong bán kính $\epsilon$ của một Core Point.
        \item \textbf{Noise Point:} Không thuộc 2 nhóm trên (gán nhãn \textbf{-1}).
      \end{itemize}
      \vspace{0.2em}
      \textbf{Cơ chế lan truyền cụm:}
      Liên thông mật độ thông qua các chuỗi điểm tiếp cận trực tiếp.
    \end{column}

\end{frame}

% Slide 15: Distribution-based - GMM & EM
\begin{frame}{5. Distribution-based: Gaussian Mixture Models (GMM)}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Mô hình xác suất tạo sinh (Generative Model):}
      $$p(x) = \sum_{k=1}^K \pi_k \mathcal{N}(x \mid \mu_k, \Sigma_k)$$
      \begin{itemize}\setlength{\itemsep}{1pt}
        \item $\pi_k$: Trọng số thành phần ($\sum \pi_k = 1$).
        \item $\mu_k$: Vector kỳ vọng (tâm cụm).
        \item $\Sigma_k$: Ma trận hiệp phương sai (hình dạng elip).
      \end{itemize}
      \textbf{Phân cụm mềm (Soft Assignment):}
      $$\gamma_{nk} = P(z_{nk}=1 \mid x_n) = \frac{\pi_k \mathcal{N}(x_n \mid \mu_k, \Sigma_k)}{\sum_{j=1}^K \pi_j \mathcal{N}(x_n \mid \mu_j, \Sigma_j)}$$
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{center}
        \includegraphics[width=0.92\linewidth,height=0.62\textheight,keepaspectratio]{images/gmm-distribution-concept.png}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

% Slide 16: Thuật toán EM và 4 Covariance Types
\begin{frame}{Thuật toán Expectation-Maximization (EM) \& Ma trận Hiệp phương sai}
  \small
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Hai bước luân phiên của EM:}
      \begin{itemize}\setlength{\itemsep}{1pt}
        \item \textbf{E-step:} Tính ma trận trách nhiệm $\gamma_{nk}$ theo tham số hiện tại.
        \item \textbf{M-step:} Cập nhật tham số theo trung bình có trọng số:
          $$\mu_k = \frac{1}{N_k}\sum \gamma_{nk}x_n, \quad \pi_k = \frac{N_k}{N}$$
          $$\Sigma_k = \frac{1}{N_k}\sum \gamma_{nk}(x_n - \mu_k)(x_n - \mu_k)^T$$
      \end{itemize}
      \textbf{Bảo đảm toán học:} Hàm Log-Likelihood luôn tăng đơn điệu sau mỗi epoch!
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{center}
        \includegraphics[width=0.95\linewidth,height=0.42\textheight,keepaspectratio]{images/gmm-covariance-types.png}
      \end{center}
      \scriptsize
      \textbf{4 Dạng ma trận hiệp phương sai:}
      \begin{itemize}\setlength{\itemsep}{0pt}
        \item \code{spherical}: Hình cầu, đẳng hướng.
        \item \code{diag}: Elip song song các trục tọa độ.
        \item \code{tied}: Các cụm chia sẻ chung 1 ma trận.
        \item \code{full}: Elip tự do mọi hướng (tổng quát).
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% Slide 17: Fuzzy Clustering - Fuzzy C-Means (FCM)
\begin{frame}{Trường phái Thứ 5: Phân cụm Mờ (Fuzzy C-Means - FCM)}
  \small
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Bản chất Phân cụm Mờ (GeeksforGeeks):}
      \begin{itemize}\setlength{\itemsep}{1pt}
        \item Mở rộng mềm từ K-Means của Bezdek (1981).
        \item Mỗi điểm thuộc về nhiều cụm với mức độ mờ $u_{ik} \in [0, 1]$ thỏa mãn $\sum_{k=1}^K u_{ik} = 1$.
      \end{itemize}
      \textbf{Hàm mục tiêu mờ ($m > 1$ là tham số mờ):}
      $$J_m = \sum_{i=1}^N \sum_{k=1}^K u_{ik}^m \|x_i - \mu_k\|^2 \to \min$$
      \textbf{Công thức cập nhật luân phiên:}
      $$u_{ik} = \frac{1}{\sum_{j=1}^K \left(\frac{\|x_i - \mu_k\|}{\|x_i - \mu_j\|}\right)^{\frac{2}{m-1}}}, \quad \mu_k = \frac{\sum_{i=1}^N u_{ik}^m x_i}{\sum_{i=1}^N u_{ik}^m}$$
    \end{column}
    \begin{column}{0.45\textwidth}
      \textbf{So sánh với K-Means \& GMM:}
      \begin{itemize}\setlength{\itemsep}{2pt}
        \item \textbf{Khi $m \to 1$:} FCM thoái hóa trở về K-Means tiêu chuẩn (Hard).
        \item \textbf{Khác biệt với GMM:} FCM không giả định phân phối Gaussian, tối ưu hóa trực tiếp trên ma trận độ thuộc mờ $u_{ik}$.
        \item \textbf{Ứng dụng tiêu biểu:} Phân đoạn ảnh MRI y tế (vùng mô não ranh giới), chấm điểm tín dụng biên.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% Slide 18: Đánh giá mô hình phân cụm
\begin{frame}{6. Đánh giá Chất lượng Phân cụm (Clustering Metrics)}
  \small
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Khi KHÔNG có nhãn thực tế (Internal):}
      \begin{itemize}\setlength{\itemsep}{1pt}
        \item \textbf{Silhouette Coefficient ($s \in [-1, 1]$):}
          $$s(i) = \frac{b(i) - a(i)}{\max(a(i), b(i))}$$
          $a(i)$: khoảng cách nội cụm trung bình; $b(i)$: khoảng cách tới cụm gần nhất. $s > 0.5 \implies$ Tốt.
        \item \textbf{Davies-Bouldin Index (DBI):} Tỷ số giữa độ phân tán nội cụm và khoảng cách liên cụm. \textbf{Càng nhỏ càng tốt}.
        \item \textbf{Calinski-Harabasz Index:} Tỷ số phương sai liên cụm / nội cụm. \textbf{Càng lớn càng tốt}.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Khi CÓ nhãn đối chuẩn (External):}
      \begin{itemize}\setlength{\itemsep}{1pt}
        \item \textbf{Adjusted Rand Index (ARI):} Đo tỷ lệ cặp điểm được gán cùng nhóm, khử yếu tố ngẫu nhiên ($-1 \le \text{ARI} \le 1$).
        \item \textbf{Normalized Mutual Information (NMI):} Tương hỗ thông tin chuẩn hóa giữa nhãn thực và nhãn dự báo ($0 \le \text{NMI} \le 1$).
      \end{itemize}
      \vspace{0.2em}
      \textbf{Chọn mô hình xác suất (GMM):}
      $$\text{BIC} = -2\ln L + p\ln N \to \min$$
      $$\text{AIC} = -2\ln L + 2p \to \min$$
    \end{column}
  \end{columns}
\end{frame}

% Slide 22: Thách thức & Hạn chế trong Phân cụm (GeeksforGeeks)
\begin{frame}{Thách thức \& Giới hạn của Phân cụm Dữ liệu (GeeksforGeeks)}
  \small
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{1. Xác định số lượng cụm $K$ tối ưu:}
      \begin{itemize}\setlength{\itemsep}{2pt}
        \item Vì không có nhãn đối chuẩn, việc chọn $K$ mang tính chủ quan.
        \item Đòi hỏi kết hợp Elbow, Silhouette và kiến thức nghiệp vụ miền (Domain Knowledge).
      \end{itemize}
      \vspace{0.4em}
      \textbf{2. Độ nhạy cảm với Khởi tạo \& Outliers:}
      \begin{itemize}\setlength{\itemsep}{2pt}
        \item K-Means dễ rơi vào cực tiểu cục bộ nếu không dùng K-Means++.
        \item Ngoại lai có thể làm sai lệch hoàn toàn vị trí tâm cụm.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{3. Lời nguyền Số chiều (Curse of Dimensionality):}
      \begin{itemize}\setlength{\itemsep}{2pt}
        \item Khi số chiều $d$ quá lớn, khoảng cách giữa các cặp điểm đều tiến về một giá trị tương đương (Distance Concentration).
        \item \textit{Giải pháp:} Cần giảm chiều bằng PCA/t-SNE trước khi gom cụm.
      \end{itemize}
      \vspace{0.4em}
      \textbf{4. Chi phí Tính toán trên Big Data:}
      \begin{itemize}\setlength{\itemsep}{2pt}
        \item Hierarchical ($O(N^2)$ bộ nhớ) bất khả thi khi $N > 20{,}000$.
        \item Cần chuyển sang Mini-Batch K-Means hoặc BIRCH.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
